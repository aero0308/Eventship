    64→# Store connected users: {sid: user_id}

    67→# Store user rooms: {user_id: set of room names}

    70→
    71→async def authenticate_socket(sid: str, environ: dict) -> Optional[str]:
    72→    """
    73→    Authenticate socket connection using JWT from query params or auth header.
    74→    Returns user_id if valid, None otherwise.
    75→    """
    76→    from jose import jwt, JWTError
    77→    from app.core.config import settings
    78→    
    79→    # Try to get token from query string or auth
    80→    query_string = environ.get('QUERY_STRING', '')
    81→    token = None
    82→    
    83→    if 'token=' in query_string:
    84→        # Parse token from query string
    85→        params = dict(p.split('=') for p in query_string.split('&') if '=' in p)
    86→        token = params.get('token')
    87→    
    88→    if not token:
    89→        # Try Authorization header
    90→        auth_header = environ.get('HTTP_AUTHORIZATION', '')
    91→        if auth_header.startswith('Bearer '):
    92→            token = auth_header[7:]
    93→    
    94→    if not token:
    95→        return None
    96→    
    97→    try:
    98→        payload = jwt.decode(token, settings.JWT_SECRET_KEY, algorithms=[settings.JWT_ALGORITHM])
    99→        return payload.get('sub')
   100→    except JWTError:
   101→        return None
   102→
   103→
   104→@sio.event
   105→async def connect(sid: str, environ: dict, auth: Optional[dict] = None):
   106→    """Handle client connection"""
   107→    user_id = await authenticate_socket(sid, environ)
   108→    
   109→    if not user_id:
   110→        print(f"❌ Socket {sid} rejected: authentication failed")
   111→        raise socketio.exceptions.ConnectionRefusedError('Authentication failed')
   112→    
   113→    connected_users[sid] = user_id
   114→    user_rooms[user_id] = set()
   115→    
   116→    # Join user's personal room
   117→    await sio.enter_room(sid, f"user_{user_id}")
   118→    user_rooms[user_id].add(f"user_{user_id}")
   119→    
   120→    print(f"✅ User {user_id} connected (sid: {sid})")
   121→    
   122→    # Send confirmation
   123→    await sio.emit('connected', {'status': 'ok', 'user_id': user_id}, to=sid)
   124→
   125→
   126→@sio.event
   127→async def disconnect(sid: str):
   128→    """Handle client disconnection"""
   129→    user_id = connected_users.pop(sid, None)
   130→    if user_id:
   131→        user_rooms.pop(user_id, None)
   132→        print(f"❌ User {user_id} disconnected (sid: {sid})")
   133→
   134→
   135→@sio.event
   136→async def join_team(sid: str, data: dict):
   137→    """Join a team room for team-wide updates"""
   138→    user_id = connected_users.get(sid)
   139→    if not user_id:
   140→        return
   141→    
   142→    team_id = data.get('team_id')
   143→    if team_id:
   144→        room = f"team_{team_id}"
   145→        await sio.enter_room(sid, room)
   146→        if user_id in user_rooms:
   147→            user_rooms[user_id].add(room)
   148→        print(f"👥 User {user_id} joined {room}")
   149→
   150→
   151→@sio.event
   152→async def leave_team(sid: str, data: dict):
   153→    """Leave a team room"""
   154→    team_id = data.get('team_id')
   155→    if team_id:
   156→        room = f"team_{team_id}"
   157→        await sio.leave_room(sid, room)
   158→        user_id = connected_users.get(sid)
   159→        if user_id and user_id in user_rooms:

   626→    private maxReconnectAttempts = 5
   627→    private listeners: Map<string, Set<Function>> = new Map()
   628→    private statusListeners: Set<(status: ConnectionStatus) => void> = new Set()
   629→    private currentStatus: ConnectionStatus = 'disconnected'
   630→    
   631→    /**
   632→     * Connect to Socket.IO server with JWT token
   633→     */
   634→    connect(token: string): void {
   635→        if (this.socket?.connected) {
   636→            console.log('Socket already connected')
   637→            return
   638→        }
   639→        
   640→        // Get base URL (strip /api/v1)
   641→        const apiUrl = import.meta.env.VITE_API_URL || 'http://localhost:8000/api/v1'
   642→        const baseUrl = apiUrl.replace('/api/v1', '')
   643→        
   644→        this.setStatus('connecting')
   645→        
   646→        this.socket = io(baseUrl, {
   647→            auth: { token },
   648→            query: { token },
   649→            transports: ['websocket', 'polling'],
   650→            reconnection: true,
   651→            reconnectionAttempts: this.maxReconnectAttempts,
   652→            reconnectionDelay: 1000,
   653→            reconnectionDelayMax: 5000
   654→        })
   655→        
   656→        this.setupListeners()
   657→    }
   658→    
   659→    /**
   660→     * Disconnect from server
   661→     */
   662→    disconnect(): void {
   663→        if (this.socket) {
   664→            this.socket.disconnect()
   665→            this.socket = null
   666→            this.setStatus('disconnected')
   667→            this.reconnectAttempts = 0
   668→        }
   669→    }
   670→    
   671→    /**
   672→     * Register event listeners
   673→     */
   674→    private setupListeners(): void {
   675→        if (!this.socket) return
   676→        
   677→        this.socket.on('connect', () => {
   678→            console.log('✅ Socket connected')
   679→            this.reconnectAttempts = 0
   680→            this.setStatus('connected')
   681→        })
   682→        
   683→        this.socket.on('disconnect', (reason) => {
   684→            console.log('❌ Socket disconnected:', reason)
   685→            this.setStatus('disconnected')
   686→            
   687→            if (reason === 'io server disconnect') {
   688→                // Server initiated disconnect - reconnect manually
   689→                this.socket?.connect()
   690→            }
   691→        })
   692→        
   693→        this.socket.on('connect_error', (error) => {
   694→            console.error('Socket connection error:', error)
   695→            this.reconnectAttempts++
   696→            
   697→            if (this.reconnectAttempts >= this.maxReconnectAttempts) {
   698→                this.setStatus('error')
   699→            } else {
   700→                this.setStatus('reconnecting')
   701→            }
   702→        })
   703→        
   704→        this.socket.io.on('reconnect_attempt', (attempt) => {
   705→            console.log(`Reconnect attempt ${attempt}`)
   706→            this.setStatus('reconnecting')
   707→        })
   708→        
   709→        this.socket.io.on('reconnect', () => {
   710→            console.log('✅ Reconnected')
   711→            this.setStatus('connected')
   712→        })
   713→        
   714→        // Forward all incoming events to registered listeners
   715→        const forwardedEvents = [
   716→            'task_updated',
   717→            'task_created',
   718→            'task_deleted',
   719→            'comment_added',
   720→            'notification',
   721→            'event_updated'
   722→        ]
   723→        
   724→        forwardedEvents.forEach(event => {
   725→            this.socket!.on(event, (data) => {
   726→                this.emit(event, data)
   727→            })
   728→        })
   729→    }
   730→    
   731→    /**
   732→     * Emit event to all registered listeners
   733→     */
   734→    private emit(event: string, data: any): void {
   735→        const listeners = this.listeners.get(event)
   736→        if (listeners) {
   737→            listeners.forEach(callback => {
   738→                try {
   739→                    callback(data)
   740→                } catch (err) {
   741→                    console.error(`Error in listener for ${event}:`, err)
   742→                }
   743→            })
   744→        }
   745→    }
   746→    
   747→    /**
   748→     * Subscribe to an event
   749→     * Returns unsubscribe function
   750→     */
   751→    on(event: string, callback: (data: any) => void): () => void {
   752→        if (!this.listeners.has(event)) {
   753→            this.listeners.set(event, new Set())
   754→        }
   755→        this.listeners.get(event)!.add(callback)
   756→        
   757→        return () => {
   758→            this.listeners.get(event)?.delete(callback)
   759→        }
   760→    }
   761→    
   762→    /**
   763→     * Subscribe to connection status changes
   764→     */
   765→    onStatusChange(callback: (status: ConnectionStatus) => void): () => void {
   766→        this.statusListeners.add(callback)
   767→        callback(this.currentStatus)
   768→        return () => this.statusListeners.delete(callback)
   769→    }
   770→    
   771→    private setStatus(status: ConnectionStatus): void {
   772→        this.currentStatus = status
   773→        this.statusListeners.forEach(cb => cb(status))
   774→    }
   775→    
   776→    /**
   777→     * Join a team room
   778→     */
   779→    joinTeam(teamId: string): void {
   780→        this.socket?.emit('join_team', { team_id: teamId })
   781→    }
   782→    
   783→    leaveTeam(teamId: string): void {
   784→        this.socket?.emit('leave_team', { team_id: teamId })
   785→    }
   786→    
   787→    /**
   788→     * Join an event room
   789→     */
   790→    joinEvent(eventId: string): void {
   791→        this.socket?.emit('join_event', { event_id: eventId })
   792→    }
   793→    
   794→    leaveEvent(eventId: string): void {
   795→        this.socket?.emit('leave_event', { event_id: eventId })
   796→    }
   797→    
   798→    getStatus(): ConnectionStatus {
   799→        return this.currentStatus
   800→    }
   801→    
   802→    isConnected(): boolean {
   803→        return this.socket?.connected || false
   804→    }
   805→}
