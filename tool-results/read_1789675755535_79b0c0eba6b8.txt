    34→/** Share target builders — resolved at click time so the URL is always current. */
    35→function shareTargets(title: string) {
    36→  return {
    37→    x: () => {
    38→      const url = encodeURIComponent(window.location.href)
    39→      const text = encodeURIComponent(title)
    40→      window.open(`https://twitter.com/intent/tweet?text=${text}&url=${url}`, '_blank', 'noopener,noreferrer')
    41→    },
    42→    linkedin: () => {
    43→      const url = encodeURIComponent(window.location.href)
    44→      window.open(`https://www.linkedin.com/sharing/share-offsite/?url=${url}`, '_blank', 'noopener,noreferrer')
    45→    },
    46→    mail: () => {
    47→      const url = window.location.href
    48→      window.location.href = `mailto:?subject=${encodeURIComponent(title)}&body=${encodeURIComponent(`I thought you might enjoy this article: ${url}`)}`
    49→    },
    50→  }
    51→}

    65→function ShareButtons({ title, layout }: { title: string; layout: 'stack' | 'row' }) {
    66→  const [copied, setCopied] = useState(false)
    67→  const timer = useRef<ReturnType<typeof setTimeout> | null>(null)
    68→
    69→  useEffect(() => () => {
    70→    if (timer.current) clearTimeout(timer.current)
    71→  }, [])
    72→
    73→  const copy = async () => {
    74→    const url = window.location.href
    75→    // Primary: async Clipboard API. Fallback: hidden-textarea execCommand for
    76→    // contexts where clipboard-write permission is restricted (older browsers,
    77→    // embedded webviews). Both failing stays silent — a copy button may lose.
    78→    try {
    79→      await navigator.clipboard.writeText(url)
    80→      setCopied(true)
    81→    } catch {
    82→      try {
    83→        const ta = document.createElement('textarea')
    84→        ta.value = url
    85→        ta.setAttribute('readonly', '')
    86→        ta.style.position = 'fixed'
    87→        ta.style.opacity = '0'
    88→        document.body.appendChild(ta)
    89→        ta.select()
    90→        document.execCommand('copy')
    91→        document.body.removeChild(ta)
    92→        setCopied(true)
    93→      } catch {
    94→        /* clipboard unavailable — no feedback shown */
    95→        return
    96→      }
    97→    }
    98→    if (timer.current) clearTimeout(timer.current)
    99→    timer.current = setTimeout(() => setCopied(false), 2200)
   100→  }
   101→
   102→  const targets = shareTargets(title)
   103→  const btn =
   104→    'flex h-10 w-10 items-center justify-center rounded-lg bg-[#1c1917] text-[#faf8f3] transition-all duration-200 hover:-translate-y-0.5 hover:bg-[#3f3a35] hover:shadow-[0_8px_20px_-8px_rgba(28,25,23,0.5)]'
   105→
   106→  return (
   107→    <div className={layout === 'row' ? 'flex items-center gap-2.5' : 'flex items-center gap-2.5'}>
   108→      <button type="button" className={btn} aria-label="Share on X (Twitter)" title="Share on X" onClick={targets.x}>
   109→        <XIcon className="h-4 w-4" />
   110→      </button>
   111→      <button type="button" className={btn} aria-label="Share on LinkedIn" title="Share on LinkedIn" onClick={targets.linkedin}>
   112→        <Linkedin aria-hidden="true" className="h-4 w-4" />
   113→      </button>
   114→      <button type="button" className={btn} aria-label="Share by email" title="Share by email" onClick={targets.mail}>
   115→        <Mail aria-hidden="true" className="h-4 w-4" />
   116→      </button>
   117→      <button
   118→        type="button"
   119→        className="flex h-10 w-10 items-center justify-center rounded-lg border border-[#e3dcd0] bg-transparent text-[#6e675f] transition-all duration-200 hover:-translate-y-0.5 hover:border-[#1c1917] hover:text-[#1c1917]"
   120→        aria-label={copied ? 'Link copied' : 'Copy link'}
   121→        title={copied ? 'Copied!' : 'Copy link'}
   122→        onClick={() => void copy()}
   123→      >
   124→        {copied ? <Check aria-hidden="true" className="h-4 w-4 text-emerald-700" /> : <Link2 aria-hidden="true" className="h-4 w-4" />}
   125→      </button>
   126→      <span aria-live="polite" className="sr-only">
   127→        {copied ? 'Article link copied to clipboard' : ''}
   128→      </span>
   129→    </div>
   130→  )
   131→}

   242→export function BlogArticlePage({ slug }: { slug: string }) {
   243→  const post = getBlogPost(slug)
   244→  const [activeId, setActiveId] = useState('')
   245→  const [progress, setProgress] = useState(0)
   246→
   247→  // Paint the document canvas cream while the article is mounted (mirrors
   248→  // HomePage's dark-canvas pattern) so overscroll never flashes the app shell.
   249→  useEffect(() => {
   250→    const previous = document.body.style.backgroundColor
   251→    document.body.style.backgroundColor = '#faf8f3'
   252→    return () => {
   253→      document.body.style.backgroundColor = previous
   254→    }
   255→  }, [])
   256→
   257→  // Fresh article = fresh scroll position. (activeId/progress re-measure via
   258→  // the scroll effect below; the AnimatePresence transition remounts per slug.)
   259→  useEffect(() => {
   260→    window.scrollTo(0, 0)
   261→  }, [slug])
   262→
   263→  // Reading progress + scroll-spy, rAF-throttled on one passive listener.
   264→  useEffect(() => {
   265→    if (!post) return
   266→    let raf = 0
   267→    const measure = () => {
   268→      const doc = document.documentElement
   269→      const total = doc.scrollHeight - window.innerHeight
   270→      setProgress(total > 0 ? Math.min(100, Math.max(0, (window.scrollY / total) * 100)) : 0)
   271→
   272→      const probe = window.scrollY + 168
   273→      let current = ''
   274→      for (const section of post.sections) {
   275→        const el = document.getElementById(section.id)
   276→        if (el && el.offsetTop <= probe) current = section.id
   277→      }
   278→      const nearBottom = window.innerHeight + window.scrollY >= doc.scrollHeight - 8
   279→      if (nearBottom && post.sections.length > 0) current = post.sections[post.sections.length - 1].id
   280→      setActiveId(current)
   281→    }
   282→    const onScroll = () => {
   283→      cancelAnimationFrame(raf)
   284→      raf = requestAnimationFrame(measure)
   285→    }
   286→    measure()
   287→    window.addEventListener('scroll', onScroll, { passive: true })
   288→    window.addEventListener('resize', onScroll)
   289→    return () => {
   290→      cancelAnimationFrame(raf)
   291→      window.removeEventListener('scroll', onScroll)
   292→      window.removeEventListener('resize', onScroll)
   293→    }
   294→  }, [post])
   295→
   296→  const scrollToSection = (id: string) => {
   297→    document.getElementById(id)?.scrollIntoView({ behavior: 'smooth', block: 'start' })
   298→  }
   299→
   300→  /** Back to the landing's blog grid (hash router owns location.hash). */
   301→  const goToBlogIndex = () => {
   302→    navigate('/')
   303→    // The route transition (AnimatePresence "wait") mounts the landing ~250ms
   304→    // after navigation, so #blog may not exist yet — retry until it does.
   305→    const tryScroll = (delay: number) => {
   306→      window.setTimeout(() => {
   307→        const el = document.getElementById('blog')
   308→        if (el) el.scrollIntoView({ behavior: 'smooth', block: 'start' })
   309→      }, delay)
   310→    }
   311→    tryScroll(420)
   312→    tryScroll(950) // safety net — re-targeting an existing smooth scroll is idempotent
   313→  }
   314→
   315→  /* ---------------- Unknown slug ---------------- */
   316→  if (!post) {
   317→    return (
   318→      <div className="blog-article flex min-h-screen flex-col items-center justify-center bg-[#faf8f3] px-6 font-fora-sans text-[#1c1917] antialiased">
   319→        <p className="font-fora-serif text-6xl italic text-[#c9c0b0]">404</p>
   320→        <h1 className="mt-4 text-xl font-semibold">This article does not exist (yet).</h1>
   321→        <p className="mt-2 max-w-sm text-center text-sm text-[#6e675f]">
   322→          The piece you are looking for may have moved, or the link is wrong.
   323→        </p>
   324→        <button
   325→          type="button"
   326→          onClick={goToBlogIndex}
   327→          className="mt-6 inline-flex min-h-11 items-center gap-2 rounded-full bg-[#1c1917] px-6 text-sm font-medium text-[#faf8f3] transition-transform hover:scale-[1.03]"
   328→        >
   329→          <ArrowLeft aria-hidden="true" className="h-4 w-4" /> Back to the blog
   330→        </button>
   331→      </div>
   332→    )
   333→  }
   334→
   335→  const related = getRelatedPosts(post.slug)
   336→
   337→  return (
   338→    <div className="blog-article min-h-screen bg-[#faf8f3] font-fora-sans text-[#1c1917] antialiased">
   339→      {/* Reading progress */}
   340→      <div aria-hidden="true" className="fixed inset-x-0 top-0 z-[95] h-[3px] bg-transparent">
   341→        <div className="h-full bg-[#1c1917] transition-[width] duration-150 ease-out" style={{ width: `${progress}%` }} />
   342→      </div>
   343→
   344→      {/* Slim header */}
   345→      <header className="mx-auto flex max-w-6xl items-center justify-between px-6 py-6">
   346→        <button
   347→          type="button"
   348→          onClick={() => navigate('/')}
   349→          className="text-sm font-bold tracking-[0.24em] text-[#1c1917] transition-opacity hover:opacity-70"
   350→        >
   351→          EVENT OS
   352→        </button>
   353→        <button
   354→          type="button"
   355→          onClick={goToBlogIndex}
   356→          className="inline-flex min-h-11 items-center gap-2 text-sm text-[#6e675f] transition-colors hover:text-[#1c1917]"
   357→        >
   358→          <ArrowLeft aria-hidden="true" className="h-4 w-4" />
   359→          Back to blog
   360→        </button>
   361→      </header>
   362→
   363→      <main className="mx-auto max-w-6xl px-6 pb-24 pt-4">
   364→        {/* Breadcrumb */}
   365→        <nav aria-label="Breadcrumb" className="flex items-center gap-1.5 text-[13px] text-[#8a8177]">
   366→          <button type="button" onClick={goToBlogIndex} className="rounded transition-colors hover:text-[#1c1917]">
   367→            Blog
   368→          </button>
   369→          <ChevronRight aria-hidden="true" className="h-3.5 w-3.5 text-[#c9c0b0]" />
   370→          <span>{post.category}</span>
   371→          <ChevronRight aria-hidden="true" className="h-3.5 w-3.5 text-[#c9c0b0]" />
   372→          <span aria-current="page" className="max-w-[180px] truncate font-medium text-[#1c1917] sm:max-w-none">
   373→            {post.title}
   374→          </span>
   375→        </nav>
   376→
   377→        {/* Hero: headline left, editorial image right */}
   378→        <div className="mt-8 grid items-start gap-10 md:grid-cols-[1fr_400px]">
   379→          <div>
   380→            <h1 className="font-fora-serif text-[clamp(2.4rem,4.8vw,3.6rem)] leading-[1.06] tracking-[-0.01em]">
   381→              {post.title}
   382→            </h1>
   383→            <div className="mt-8 flex flex-wrap items-center gap-x-3 gap-y-2">
   384→              <Image
   385→                src={post.author.avatar}
   386→                alt={`Portrait of ${post.author.name}`}
   387→                width={40}

   389→                unoptimized
   390→                className="h-10 w-10 rounded-full object-cover ring-1 ring-[#e3dcd0]"
   391→              />
   392→              <span className="text-sm font-medium">{post.author.name}</span>

   394→              <span className="text-sm text-[#8a8177]">Last updated on {post.updatedOn}</span>
   395→              <span aria-hidden="true" className="h-4 w-px bg-[#ddd5c7]" />

   441→          {/* Article */}
