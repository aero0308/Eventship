   156→   156→import { useState, useRef, useEffect } from 'react'
   157→   157→
   158→   158→interface DropdownItem {
   159→   159→    label: string
   160→   160→    onClick: () => void
   161→   161→    icon?: React.ReactNode
   162→   162→    variant?: 'default' | 'danger'
   163→   163→    disabled?: boolean
   164→   164→}
   165→   165→
   166→   166→interface DropdownProps {
   167→   167→    trigger: React.ReactNode
   168→   168→    items: DropdownItem[]
   169→   169→    align?: 'left' | 'right'
   170→   170→}
   171→   171→
   172→   172→export const Dropdown = ({ trigger, items, align = 'right' }: DropdownProps) => {
   173→   173→    const [isOpen, setIsOpen] = useState(false)
   174→   174→    const dropdownRef = useRef<HTMLDivElement>(null)
   175→   175→    
   176→   176→    useEffect(() => {
   177→   177→        const handleClickOutside = (e: MouseEvent) => {
   178→   178→            if (dropdownRef.current && !dropdownRef.current.contains(e.target as Node)) {
   179→   179→                setIsOpen(false)
   180→   180→            }
   181→   181→        }
   182→   182→        document.addEventListener('mousedown', handleClickOutside)
   183→   183→        return () => document.removeEventListener('mousedown', handleClickOutside)
   184→   184→    }, [])
   185→   185→    
   186→   186→    return (
   187→   187→        <div className="relative" ref={dropdownRef}>
   188→   188→            <div onClick={() => setIsOpen(!isOpen)}>{trigger}</div>
   189→   189→            {isOpen && (
   190→   190→                <div className={`absolute z-50 mt-2 w-48 bg-white rounded-lg shadow-lg border border-gray-200 py-1 ${align === 'right' ? 'right-0' : 'left-0'}`}>
   191→   191→                    {items.map((item, idx) => (
   192→   192→                        <button
   193→   193→                            key={idx}
   194→   194→                            onClick={() => { item.onClick(); setIsOpen(false) }}
   195→   195→                            disabled={item.disabled}
   196→   196→                            className={`w-full text-left px-4 py-2 text-sm hover:bg-gray-50 disabled:opacity-50 ${item.variant === 'danger' ? 'text-red-600' : 'text-gray-700'}`}
   197→   197→                        >
   198→   198→                            {item.icon && <span className="mr-2">{item.icon}</span>}
   199→   199→                            {item.label}
   200→   200→                        </button>
   201→   201→                    ))}
   202→   202→                </div>
   203→   203→            )}
   204→   204→        </div>
   205→   205→    )
