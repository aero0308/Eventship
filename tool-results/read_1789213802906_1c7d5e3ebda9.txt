   712→/**
   713→ * Extract error message from various error types
   714→ */
   715→export const getErrorMessage = (error: unknown): string => {
   716→    if (error instanceof Error) {
   717→        return error.message
   718→    }
   719→    
   720→    if (typeof error === 'string') {
   721→        return error
   722→    }
   723→    
   724→    if (error && typeof error === 'object' && 'detail' in error) {
   725→        const detail = (error as any).detail
   726→        if (typeof detail === 'string') return detail
   727→        if (Array.isArray(detail)) {
   728→            return detail.map((d: any) => d.msg || d.message).join(', ')
   729→        }
   730→    }
   731→    
   732→    return 'Something went wrong. Please try again.'
   733→}
   734→
   735→/**
   736→ * Extract field-specific errors from validation errors
   737→ */
   738→export const getFieldErrors = (error: unknown): Record<string, string> => {
   739→    const errors: Record<string, string> = {}
   740→    
   741→    if (error && typeof error === 'object' && 'response' in error) {
   742→        const axiosError = error as AxiosError
   743→        const data = axiosError.response?.data as any
   744→        
   745→        if (Array.isArray(data?.detail)) {
   746→            data.detail.forEach((err: any) => {
   747→                if (err.loc && err.loc.length > 1) {
   748→                    const field = err.loc[err.loc.length - 1]
   749→                    errors[field] = err.msg
   750→                }
   751→            })
   752→        }
   753→    }
   754→    
   755→    return errors
   756→}

  1239→    action: () => void
  1240→    description: string

  1243→export const useKeyboardShortcuts = (shortcuts: Shortcut[]) => {
  1244→    const navigate = useNavigate()

  1247→        const handleKeyDown = (e: KeyboardEvent) => {
  1248→            // Skip if typing in input
  1249→            const target = e.target as HTMLElement
  1250→            if (['INPUT', 'TEXTAREA', 'SELECT'].includes(target.tagName)) return
  1251→            
  1252→            shortcuts.forEach(shortcut => {
  1253→                const keyMatch = e.key.toLowerCase() === shortcut.key.toLowerCase()
  1254→                const ctrlMatch = shortcut.ctrl ? (e.ctrlKey || e.metaKey) : !(e.ctrlKey || e.metaKey)
  1255→                const shiftMatch = shortcut.shift ? e.shiftKey : !e.shiftKey
  1256→                const altMatch = shortcut.alt ? e.altKey : !e.altKey
  1257→                
  1258→                if (keyMatch && ctrlMatch && shiftMatch && altMatch) {
  1259→                    e.preventDefault()
  1260→                    shortcut.action()
  1261→                }
  1262→            })
  1263→        }
  1264→        
  1265→        window.addEventListener('keydown', handleKeyDown)
  1266→        return () => window.removeEventListener('keydown', handleKeyDown)
  1267→    }, [shortcuts])

  1315→

  1321→            'button, [href], input, select, textarea, [tabindex]:not([tabindex="-1"])'
  1322→        )

  1324→        const lastElement = focusableElements[focusableElements.length - 1] as HTMLElement
  1325→

  1334→                    e.preventDefault()
  1335→                    lastElement?.focus()
  1336→                }
  1337→            } else {
  1338→                if (document.activeElement === lastElement) {
  1339→                    e.preventDefault()
  1340→                    firstElement?.focus()
  1341→                }
  1342→            }
  1343→        }
  1344→        
  1345→        container.addEventListener('keydown', handleTabKey)

  1628→
  1629→export function useDebounce<T>(value: T, delay: number = 300): T {
  1630→    const [debouncedValue, setDebouncedValue] = useState<T>(value)
  1631→    
  1632→    useEffect(() => {
  1633→        const handler = setTimeout(() => setDebouncedValue(value), delay)
  1634→        return () => clearTimeout(handler)
  1635→    }, [value, delay])
  1636→    
  1637→    return debouncedValue
  1638→}
