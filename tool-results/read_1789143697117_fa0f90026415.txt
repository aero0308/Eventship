    83→function teamTone(name: string): string {
    84→  let hash = 0
    85→  for (let i = 0; i < name.length; i += 1) hash = (hash * 31 + name.charCodeAt(i)) % 997
    86→  return TEAM_TONES[hash % TEAM_TONES.length]
    87→}

    89→/** Completion pill tone: emerald when mostly done, amber when started, neutral at 0. */
    90→function completionPill(percent: number): string {
    91→  if (percent >= 75) return 'bg-emerald-100 text-emerald-800 dark:bg-emerald-500/15 dark:text-emerald-300'
    92→  if (percent > 0) return 'bg-amber-100 text-amber-800 dark:bg-amber-500/15 dark:text-amber-300'
    93→  return 'bg-muted text-muted-foreground'
    94→}

   126→function focusDueLabel(dueDate: string | null): string {
   127→  if (!dueDate) return 'No due date'
   128→  const diff = differenceInCalendarDays(new Date(dueDate), new Date())
   129→  if (diff === 0) return 'Due today'
   130→  if (diff === 1) return 'Due tomorrow'
   131→  if (diff < 0) return `Overdue by ${Math.abs(diff)}d`
   132→  return `Due ${format(new Date(dueDate), 'EEE, MMM d')}`
   133→}

   300→export function DashboardPage() {
   301→  const user = useAuthStore((s) => s.user)
   302→  const [stats, setStats] = useState<DashboardStatsDTO | null>(null)
   303→  const [loading, setLoading] = useState(true)
   304→
   305→  // Focus strip scope — "All tasks" or "Mine only". Persisted per browser.
   306→  const [focusScope, setFocusScope] = useState<'all' | 'mine'>(() => {
   307→    if (typeof window === 'undefined') return 'all'
   308→    return window.localStorage.getItem('ems-focus-scope') === 'mine' ? 'mine' : 'all'
   309→  })
   310→  const [focusPending, setFocusPending] = useState(false)
   311→
   312→  useEffect(() => {
   313→    let cancelled = false
   314→    api
   315→      .get<{ stats: DashboardStatsDTO }>(`/dashboard${focusScope === 'mine' ? '?focus=mine' : ''}`)
   316→      .then((data) => {
   317→        if (!cancelled) setStats(data.stats)
   318→      })
   319→      .catch(() => {
   320→        // Rendered empty state handles null stats below.
   321→      })
   322→      .finally(() => {
   323→        if (!cancelled) {
   324→          setLoading(false)
   325→          setFocusPending(false)
   326→        }
   327→      })
   328→    return () => {
   329→      cancelled = true
   330→    }
   331→  }, [focusScope])
   332→
   333→  const handleFocusScopeChange = (scope: 'all' | 'mine') => {
   334→    if (scope === focusScope) return
   335→    setFocusScope(scope)
   336→    setFocusPending(true)
   337→    try {
   338→      window.localStorage.setItem('ems-focus-scope', scope)
   339→    } catch {
   340→      // Storage unavailable — the toggle still works for this session.
   341→    }
   342→  }
   343→
   344→  const statusData = useMemo(
   345→    () => (stats?.tasksByStatus ?? []).map((entry) => ({ name: TASK_STATUS_LABELS[entry.status] ?? entry.status, count: entry.count, key: entry.status })),
   346→    [stats]
   347→  )
   348→
   349→  const priorityData = useMemo(
   350→    () => (stats?.tasksByPriority ?? []).map((entry) => ({ name: PRIORITY_LABELS[entry.priority] ?? entry.priority, value: entry.count, key: entry.priority })),
   351→    [stats]
   352→  )
   353→
   354→  const openTasks = stats ? stats.totals.tasks - stats.totals.completedTasks : 0
   355→
   356→  return (
   357→    <div>
   358→      <PageHeader
   359→        title="Dashboard"
   360→        subtitle={user ? `${greeting()}, ${user.fullName} — here's what's happening across your events.` : "Here's what's happening across your events."}
   361→      />
   362→
   363→      {/* ============ My focus strip ============ */}
   364→      {loading ? (
   365→        <Skeleton className="h-20 rounded-2xl" />
   366→      ) : stats ? (
   367→        <div className="mt-1 mb-4">
   368→          <FocusStrip
   369→            myFocus={stats.myFocus}
   370→            scope={focusScope}
   371→            pending={focusPending}
   372→            onScopeChange={handleFocusScopeChange}
   373→          />
   374→        </div>
   375→      ) : null}
   376→
   377→      {/* ============ Stat cards ============ */}
   378→      <section className="grid grid-cols-1 gap-4 sm:grid-cols-2 xl:grid-cols-4" aria-label="Key statistics">
   379→        {loading || !stats ? (
   380→          Array.from({ length: 4 }).map((_, i) => <Skeleton key={i} className="h-32 rounded-xl" />)
   381→        ) : (
   382→          <>
   383→            <StatCard
   384→              icon={CalendarDays}
   385→              label="Active Events"
   386→              value={stats.totals.activeEvents}
   387→              sub={`${stats.totals.events} events in total`}
   388→              tint="emerald"
   389→            />
   390→            <StatCard
   391→              icon={ListTodo}
   392→              label="Open Tasks"
   393→              value={openTasks}
   394→              sub={`${stats.totals.tasks} tasks tracked overall`}
   395→              tint="amber"
   396→            />
   397→            <StatCard
   398→              icon={CheckCircle2}
   399→              label="Completed"
   400→              value={stats.totals.completedTasks}
   401→              sub={`${Math.round(stats.completionRate)}% completion rate`}
   402→              tint="stone"
   403→              progress={stats.completionRate}
   404→            />
   405→            <StatCard
   406→              icon={AlertTriangle}
   407→              label="Blocked"
   408→              value={stats.totals.blockedTasks}
   409→              sub={stats.totals.blockedTasks > 0 ? 'Needs immediate attention' : 'Nothing blocked — nice!'}
   410→              tint="red"
   411→            />
   412→          </>
   413→        )}
   414→      </section>
   415→
   416→      {/* ============ Charts ============ */}
   417→      <section className="mt-6 grid grid-cols-1 gap-4 lg:grid-cols-2" aria-label="Task charts">
