   158→def get_dashboard_stats(db: Session, user: Optional[User] = None) -> Dict:
   159→    """
   160→    Get overall dashboard statistics.
   161→    
   162→    Authorization:
   163→    - EVENT_MANAGER: sees all data
   164→    - TEAM_LEADER: sees team data
   165→    - EMPLOYEE: sees only their tasks
   166→    """
   167→    # Base queries with role-based filtering
   168→    if user and user.role == UserRole.EMPLOYEE:
   169→        # Employee: only their tasks
   170→        task_query = db.query(Task).filter(Task.assigned_to == user.id)
   171→        event_query = db.query(Event).join(Team).filter(Team.id == user.team_id)
   172→        team_query = db.query(Team).filter(Team.id == user.team_id)
   173→        user_query = db.query(User).filter(User.team_id == user.team_id)
   174→    elif user and user.role == UserRole.TEAM_LEADER:
   175→        # Team leader: team's data
   176→        task_query = db.query(Task).join(Event).filter(Event.team_id == user.team_id)
   177→        event_query = db.query(Event).filter(Event.team_id == user.team_id)
   178→        team_query = db.query(Team).filter(Team.id == user.team_id)
   179→        user_query = db.query(User).filter(User.team_id == user.team_id)
   180→    else:
   181→        # Manager: all data
   182→        task_query = db.query(Task)
   183→        event_query = db.query(Event)
   184→        team_query = db.query(Team)
   185→        user_query = db.query(User)
   186→    
   187→    # Count stats
   188→    total_tasks = task_query.count()
   189→    completed_tasks = task_query.filter(Task.status == TaskStatus.COMPLETED).count()
   190→    in_progress_tasks = task_query.filter(Task.status == TaskStatus.IN_PROGRESS).count()
   191→    blocked_tasks = task_query.filter(Task.status == TaskStatus.BLOCKED).count()
   192→    not_started_tasks = task_query.filter(Task.status == TaskStatus.NOT_STARTED).count()
   193→    overdue_tasks = task_query.filter(
   194→        Task.due_date < datetime.utcnow(),
   195→        Task.status != TaskStatus.COMPLETED
   196→    ).count()
   197→    
   198→    total_events = event_query.count()
   199→    active_events = event_query.filter(
   200→        Event.status.in_([EventStatus.PLANNING, EventStatus.IN_PROGRESS])
   201→    ).count()
   202→    
   203→    total_teams = team_query.count()
   204→    total_users = user_query.count()
   205→    
   206→    completion_rate = (completed_tasks / total_tasks * 100) if total_tasks > 0 else 0
   207→    
   208→    return {
   209→        "total_events": total_events,
   210→        "active_events": active_events,
   211→        "total_tasks": total_tasks,
   212→        "completed_tasks": completed_tasks,
   213→        "in_progress_tasks": in_progress_tasks,
   214→        "blocked_tasks": blocked_tasks,
   215→        "not_started_tasks": not_started_tasks,
   216→        "overdue_tasks": overdue_tasks,
   217→        "completion_rate": round(completion_rate, 2),
   218→        "total_teams": total_teams,
   219→        "total_users": total_users
   220→    }

   301→    """Get all blocked tasks with details"""

   321→        
   322→        result.append({
   323→            "id": t.id,
   324→            "title": t.title,
   325→            "priority": t.priority.value,

   328→            "due_date": t.due_date,
   329→            "days_blocked": days_blocked,

   453→def get_progress_over_time(
   454→    db: Session,
   455→    user: Optional[User] = None,
   456→    days: int = 30
   457→) -> List[Dict]:
   458→    """Get daily progress data for the last N days"""
   459→    result = []
   460→    now = datetime.utcnow()
   461→    
   462→    for i in range(days, -1, -1):
   463→        date = now - timedelta(days=i)
   464→        date_start = date.replace(hour=0, minute=0, second=0, microsecond=0)
   465→        date_end = date_start + timedelta(days=1)
   466→        
   467→        query_created = db.query(Task).filter(
   468→            Task.created_at.between(date_start, date_end)
   469→        )
   470→        query_completed = db.query(Task).filter(
   471→            Task.status == TaskStatus.COMPLETED,
   472→            Task.updated_at.between(date_start, date_end)
   473→        )
   474→        
   475→        if user:
   476→            if user.role == UserRole.EMPLOYEE:
   477→                query_created = query_created.filter(Task.assigned_to == user.id)
   478→                query_completed = query_completed.filter(Task.assigned_to == user.id)
   479→            elif user.role == UserRole.TEAM_LEADER:
   480→                query_created = query_created.join(Event).filter(Event.team_id == user.team_id)
   481→                query_completed = query_completed.join(Event).filter(Event.team_id == user.team_id)
   482→        
   483→        result.append({
   484→            "date": date_start.strftime("%Y-%m-%d"),
   485→            "created": query_created.count(),
   486→            "completed": query_completed.count(),
   487→            "cumulative_completed": 0  # Will calculate below
   488→        })
   489→    
   490→    # Calculate cumulative
   491→    cumulative = 0
   492→    for item in result:
   493→        cumulative += item["completed"]
   494→        item["cumulative_completed"] = cumulative
   495→    
   496→    return result
