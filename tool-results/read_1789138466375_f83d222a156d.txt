   135→function focusDueLabel(dueDate: string | null): string {
   136→  if (!dueDate) return 'No due date'
   137→  const diff = differenceInCalendarDays(new Date(dueDate), new Date())
   138→  if (diff === 0) return 'Due today'
   139→  if (diff === 1) return 'Due tomorrow'
   140→  if (diff < 0) return `Overdue by ${Math.abs(diff)}d`
   141→  return `Due ${format(new Date(dueDate), 'EEE, MMM d')}`
   142→}

   309→export function DashboardPage() {
   310→  const user = useAuthStore((s) => s.user)
   311→  const [stats, setStats] = useState<DashboardStatsDTO | null>(null)
   312→  const [loading, setLoading] = useState(true)
   313→
   314→  // Focus strip scope — "All tasks" or "Mine only". Persisted per browser.
   315→  const [focusScope, setFocusScope] = useState<'all' | 'mine'>(() => {
   316→    if (typeof window === 'undefined') return 'all'
   317→    return window.localStorage.getItem('ems-focus-scope') === 'mine' ? 'mine' : 'all'
   318→  })
   319→  const [focusPending, setFocusPending] = useState(false)
   320→
   321→  useEffect(() => {
   322→    let cancelled = false
   323→    api
   324→      .get<{ stats: DashboardStatsDTO }>(`/dashboard${focusScope === 'mine' ? '?focus=mine' : ''}`)
   325→      .then((data) => {
   326→        if (!cancelled) setStats(data.stats)
   327→      })
   328→      .catch(() => {
   329→        // Rendered empty state handles null stats below.
   330→      })
   331→      .finally(() => {
   332→        if (!cancelled) {
   333→          setLoading(false)
   334→          setFocusPending(false)
   335→        }
   336→      })
   337→    return () => {
   338→      cancelled = true
   339→    }
   340→  }, [focusScope])
   341→
   342→  const handleFocusScopeChange = (scope: 'all' | 'mine') => {
   343→    if (scope === focusScope) return
   344→    setFocusScope(scope)
   345→    setFocusPending(true)
   346→    try {
   347→      window.localStorage.setItem('ems-focus-scope', scope)
   348→    } catch {
   349→      // Storage unavailable — the toggle still works for this session.
   350→    }
   351→  }
   352→
   353→  const statusData = useMemo(
   354→    () => (stats?.tasksByStatus ?? []).map((entry) => ({ name: TASK_STATUS_LABELS[entry.status] ?? entry.status, count: entry.count, key: entry.status })),
   355→    [stats]
   356→  )
   357→
   358→  const priorityData = useMemo(
   359→    () => (stats?.tasksByPriority ?? []).map((entry) => ({ name: PRIORITY_LABELS[entry.priority] ?? entry.priority, value: entry.count, key: entry.priority })),
   360→    [stats]
   361→  )
   362→
   363→  const openTasks = stats ? stats.totals.tasks - stats.totals.completedTasks : 0
   364→
   365→  return (
   366→    <div>
   367→      <PageHeader
   368→        title="Dashboard"
   369→        subtitle={user ? `${greeting()}, ${user.fullName} — here's what's happening across your events.` : "Here's what's happening across your events."}
   370→      />
   371→
   372→      {/* ============ My focus strip ============ */}
   373→      {loading ? (
   374→        <Skeleton className="h-20 rounded-2xl" />
   375→      ) : stats ? (
   376→        <div className="mt-1 mb-4">
   377→          <FocusStrip
   378→            myFocus={stats.myFocus}
   379→            scope={focusScope}
   380→            pending={focusPending}
   381→            onScopeChange={handleFocusScopeChange}
   382→          />
   383→        </div>
   384→      ) : null}
   385→
   386→      {/* ============ Stat cards ============ */}
   387→      <section className="grid grid-cols-1 gap-4 sm:grid-cols-2 xl:grid-cols-4" aria-label="Key statistics">
   388→        {loading || !stats ? (
   389→          Array.from({ length: 4 }).map((_, i) => <Skeleton key={i} className="h-32 rounded-xl" />)
   390→        ) : (
   391→          <>
   392→            <StatCard
   393→              icon={CalendarDays}
   394→              label="Active Events"
   395→              value={stats.totals.activeEvents}
   396→              sub={`${stats.totals.events} events in total`}
   397→              tint="emerald"
   398→            />
   399→            <StatCard
   400→              icon={ListTodo}
   401→              label="Open Tasks"
   402→              value={openTasks}
   403→              sub={`${stats.totals.tasks} tasks tracked overall`}
   404→              tint="amber"
   405→            />
   406→            <StatCard
   407→              icon={CheckCircle2}
   408→              label="Completed"
   409→              value={stats.totals.completedTasks}
   410→              sub={`${Math.round(stats.completionRate)}% completion rate`}
   411→              tint="stone"
   412→              progress={stats.completionRate}
   413→            />
   414→            <StatCard
   415→              icon={AlertTriangle}
   416→              label="Blocked"
   417→              value={stats.totals.blockedTasks}
   418→              sub={stats.totals.blockedTasks > 0 ? 'Needs immediate attention' : 'Nothing blocked — nice!'}
   419→              tint="red"
   420→            />
   421→          </>
   422→        )}
   423→      </section>
   424→
   425→      {/* ============ Charts ============ */}
   426→      <section className="mt-6 grid grid-cols-1 gap-4 lg:grid-cols-2" aria-label="Task charts">
   427→        <Card>
   428→          <CardHeader>
   429→            <CardTitle className="text-base">Tasks by status</CardTitle>

   432→            {loading ? (

   435→              <EmptyState icon={ListTodo} title="No tasks yet" hint="Create tasks to see the status breakdown here." />
   436→            ) : (
   437→              <div className="h-64 w-full">
   438→                <ResponsiveContainer width="100%" height="100%">
