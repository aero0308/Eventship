    71→    71→async def authenticate_socket(sid: str, environ: dict) -> Optional[str]:
    72→    72→    """
    73→    73→    Authenticate socket connection using JWT from query params or auth header.
    74→    74→    Returns user_id if valid, None otherwise.
    75→    75→    """
    76→    76→    from jose import jwt, JWTError
    77→    77→    from app.core.config import settings
    78→    78→    
    79→    79→    # Try to get token from query string or auth
    80→    80→    query_string = environ.get('QUERY_STRING', '')
    81→    81→    token = None
    82→    82→    
    83→    83→    if 'token=' in query_string:
    84→    84→        # Parse token from query string
    85→    85→        params = dict(p.split('=') for p in query_string.split('&') if '=' in p)
    86→    86→        token = params.get('token')
    87→    87→    
    88→    88→    if not token:
    89→    89→        # Try Authorization header
    90→    90→        auth_header = environ.get('HTTP_AUTHORIZATION', '')
    91→    91→        if auth_header.startswith('Bearer '):
    92→    92→            token = auth_header[7:]
    93→    93→    
    94→    94→    if not token:
    95→    95→        return None
    96→    96→    
    97→    97→    try:
    98→    98→        payload = jwt.decode(token, settings.JWT_SECRET_KEY, algorithms=[settings.JWT_ALGORITHM])
    99→    99→        return payload.get('sub')
   100→   100→    except JWTError:
   101→   101→        return None

   735→   735→        const listeners = this.listeners.get(event)
   736→   736→        if (listeners) {
   737→   737→            listeners.forEach(callback => {
   738→   738→                try {
   739→   739→                    callback(data)
   740→   740→                } catch (err) {

   742→   742→                }
   743→   743→            })

   749→   749→     * Returns unsubscribe function
   750→   750→     */
   751→   751→    on(event: string, callback: (data: any) => void): () => void {
   752→   752→        if (!this.listeners.has(event)) {
   753→   753→            this.listeners.set(event, new Set())
   754→   754→        }
   755→   755→        this.listeners.get(event)!.add(callback)
   756→   756→        
   757→   757→        return () => {
   758→   758→            this.listeners.get(event)?.delete(callback)
   759→   759→        }
   760→   760→    }
