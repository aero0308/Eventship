    83→    83→function teamTone(name: string): string {
    84→    84→  let hash = 0
    85→    85→  for (let i = 0; i < name.length; i += 1) hash = (hash * 31 + name.charCodeAt(i)) % 997
    86→    86→  return TEAM_TONES[hash % TEAM_TONES.length]
    87→    87→}

    90→    90→function completionPill(percent: number): string {

   126→   126→function focusDueLabel(dueDate: string | null): string {
   127→   127→  if (!dueDate) return 'No due date'
   128→   128→  const diff = differenceInCalendarDays(new Date(dueDate), new Date())
   129→   129→  if (diff === 0) return 'Due today'
   130→   130→  if (diff === 1) return 'Due tomorrow'
   131→   131→  if (diff < 0) return `Overdue by ${Math.abs(diff)}d`
   132→   132→  return `Due ${format(new Date(dueDate), 'EEE, MMM d')}`
   133→   133→}

   304→   304→
   305→   305→  // Focus strip scope — "All tasks" or "Mine only". Persisted per browser.
   306→   306→  const [focusScope, setFocusScope] = useState<'all' | 'mine'>(() => {
   307→   307→    if (typeof window === 'undefined') return 'all'
   308→   308→    return window.localStorage.getItem('ems-focus-scope') === 'mine' ? 'mine' : 'all'
   309→   309→  })
   310→   310→  const [focusPending, setFocusPending] = useState(false)
   311→   311→
   312→   312→  useEffect(() => {
   313→   313→    let cancelled = false
   314→   314→    api
   315→   315→      .get<{ stats: DashboardStatsDTO }>(`/dashboard${focusScope === 'mine' ? '?focus=mine' : ''}`)
   316→   316→      .then((data) => {
   317→   317→        if (!cancelled) setStats(data.stats)
   318→   318→      })
   319→   319→      .catch(() => {
   320→   320→        // Rendered empty state handles null stats below.
   321→   321→      })
   322→   322→      .finally(() => {
   323→   323→        if (!cancelled) {
   324→   324→          setLoading(false)
   325→   325→          setFocusPending(false)
   326→   326→        }
   327→   327→      })
   328→   328→    return () => {
   329→   329→      cancelled = true
   330→   330→    }
   331→   331→  }, [focusScope])
   332→   332→
   333→   333→  const handleFocusScopeChange = (scope: 'all' | 'mine') => {
   334→   334→    if (scope === focusScope) return
   335→   335→    setFocusScope(scope)
   336→   336→    setFocusPending(true)
   337→   337→    try {
   338→   338→      window.localStorage.setItem('ems-focus-scope', scope)
   339→   339→    } catch {
   340→   340→      // Storage unavailable — the toggle still works for this session.
   341→   341→    }
   342→   342→  }
   343→   343→
   344→   344→  const statusData = useMemo(
   345→   345→    () => (stats?.tasksByStatus ?? []).map((entry) => ({ name: TASK_STATUS_LABELS[entry.status] ?? entry.status, count: entry.count, key: entry.status })),

   356→   356→  return (

   361→   361→      />
   362→   362→
   363→   363→      {/* ============ My focus strip ============ */}
   364→   364→      {loading ? (

   367→   367→        <div className="mt-1 mb-4">
   368→   368→          <FocusStrip
   369→   369→            myFocus={stats.myFocus}
   370→   370→            scope={focusScope}
   371→   371→            pending={focusPending}
   372→   372→            onScopeChange={handleFocusScopeChange}
   373→   373→          />
   374→   374→        </div>
   375→   375→      ) : null}
   376→   376→
   377→   377→      {/* ============ Stat cards ============ */}
   378→   378→      <section className="grid grid-cols-1 gap-4 sm:grid-cols-2 xl:grid-cols-4" aria-label="Key statistics">
   379→   379→        {loading || !stats ? (
   380→   380→          Array.from({ length: 4 }).map((_, i) => <Skeleton key={i} className="h-32 rounded-xl" />)
   381→   381→        ) : (
   382→   382→          <>
   383→   383→            <StatCard
   384→   384→              icon={CalendarDays}
   385→   385→              label="Active Events"
   386→   386→              value={stats.totals.activeEvents}

   398→   398→              icon={CheckCircle2}
   399→   399→              label="Completed"
   400→   400→              value={stats.totals.completedTasks}
   401→   401→              sub={`${Math.round(stats.completionRate)}% completion rate`}
   402→   402→              tint="stone"
   403→   403→              progress={stats.completionRate}
   404→   404→            />
   405→   405→            <StatCard
   406→   406→              icon={AlertTriangle}
   407→   407→              label="Blocked"
   408→   408→              value={stats.totals.blockedTasks}
   409→   409→              sub={stats.totals.blockedTasks > 0 ? 'Needs immediate attention' : 'Nothing blocked — nice!'}
   410→   410→              tint="red"
   411→   411→            />

   416→   416→      {/* ============ Charts ============ */}
   417→   417→      <section className="mt-6 grid grid-cols-1 gap-4 lg:grid-cols-2" aria-label="Task charts">
   418→   418→        <Card>

   420→   420→            <CardTitle className="text-base">Tasks by status</CardTitle>

   422→   422→          <CardContent>
