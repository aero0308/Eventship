   113→function dueChip(task: TaskDTO): { label: string; classes: string } | null {
   114→  if (!task.dueDate) return null
   115→  const due = new Date(task.dueDate)
   116→  const days = differenceInCalendarDays(due, new Date())
   117→  const done = task.status === 'COMPLETED'
   118→  if (done) {
   119→    return { label: `Due ${format(due, 'MMM d')}`, classes: 'border-border bg-muted/50 text-muted-foreground/70' }
   120→  }
   121→  if (days < 0) return { label: `${Math.abs(days)}d overdue`, classes: 'border-red-200 bg-red-50 text-red-700' }
   122→  if (days === 0) return { label: 'Due today', classes: 'border-amber-200 bg-amber-50 text-amber-800' }
   123→  if (days < 3) return { label: `${days}d left`, classes: 'border-amber-200 bg-amber-50 text-amber-800' }
   124→  return { label: `Due ${format(due, 'MMM d')}`, classes: 'border-border bg-muted/50 text-muted-foreground' }
   125→}

   136→function DraggableTaskCard({ task, canDrag, mobileStatusSelect, onOpen }: TaskCardProps) {

   139→  const assignee = task.assignee
   140→
   141→  return (
   142→    <div ref={setNodeRef} style={{ transform: CSS.Translate.toString(transform) }} className={cn(isDragging && 'z-20 opacity-60')}>
   143→      <Card
   144→        className={cn(
   145→          'cursor-pointer gap-2 border-l-4 py-3 shadow-sm transition-shadow hover:shadow-md',
   146→          'border-l-red-500',
   147→          task.priority === 'MEDIUM' && 'border-l-amber-500',
   148→          task.priority === 'LOW' && 'border-l-stone-300',
   149→          task.status === 'COMPLETED' && 'opacity-80',
   150→          isDragging && 'ring-2 ring-emerald-500',

   152→        )}

   155→        tabIndex={0}

   161→        }}
   162→        aria-label={`Open task ${task.title}`}
   163→      >
   164→        <CardContent className="px-3" {...(canDrag ? listeners : {})} {...attributes}>
   165→          <div className="flex items-start justify-between gap-2">
   166→            <p className={cn('line-clamp-2 text-sm font-semibold text-foreground', task.status === 'COMPLETED' && 'line-through decoration-stone-300')}>
