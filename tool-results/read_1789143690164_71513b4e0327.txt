    71→function initialsOfName(name: string): string {
    72→  return name
    73→    .split(/\s+/)
    74→    .filter(Boolean)
    75→    .slice(0, 2)
    76→    .map((part) => part[0]?.toUpperCase() ?? '')
    77→    .join('')
    78→}

   104→function formatRange(startISO: string, endISO: string): string {
   105→  const start = new Date(startISO)
   106→  const end = new Date(endISO)
   107→  if (start.getFullYear() === end.getFullYear()) {
   108→    if (format(start, 'MMM d, yyyy') === format(end, 'MMM d, yyyy')) return format(start, 'MMM d, yyyy')
   109→    return `${format(start, 'MMM d')} – ${format(end, 'MMM d, yyyy')}`
   110→  }
   111→  return `${format(start, 'MMM d, yyyy')} – ${format(end, 'MMM d, yyyy')}`
   112→}

   134→  useEffect(() => {
   135→    if (!user) return

   142→      const summaries = await fetchPresenceSummary(['event:'])

   145→        const next: Record<string, PresenceUser[]> = { ...prev }

   147→        for (const { room, viewers } of summaries) {
   148→          const eventId = room.slice('event:'.length)
   149→          const others = viewers.filter((v) => v.id !== user.id)
   150→          if (others.length > 0) next[eventId] = others
   151→          else delete next[eventId]
   152→        }
   153→        return next
   154→      })

   157→    const onGlobal = (payload: PresenceGlobalPayload) => {
   158→      if (!payload?.room?.startsWith('event:')) return

   160→      const others = payload.viewers.filter((v) => v.id !== user.id)

   162→        const next = { ...prev }
   163→        if (others.length > 0) next[eventId] = others
   164→        else delete next[eventId]
   165→        return next
   166→      })
   167→    }
   168→
   169→    socket.on('presence:global', onGlobal)
   170→    socket.once('connect', snapshot)
   171→    if (socket.connected) void snapshot()
   172→    return () => {
   173→      disposed = true
   174→      socket.off('presence:global', onGlobal)
   175→      socket.off('connect', snapshot)

   183→  const [saving, setSaving] = useState(false)
   184→

   368→            />
   369→          </div>
   370→          <Select value={statusFilter} onValueChange={setStatusFilter}>
   371→            <SelectTrigger className="h-11 w-full sm:w-44" aria-label="Filter by status">
   372→              <SelectValue placeholder="All statuses" />
   373→            </SelectTrigger>
   374→            <SelectContent>
   375→              <SelectItem value="all">All statuses</SelectItem>

   377→                <SelectItem key={status} value={status}>
   378→                  {EVENT_STATUS_LABELS[status]}

   396→        </div>
   397→
   398→        <div className="flex flex-col gap-3 sm:flex-row sm:items-center">

   400→            className="flex items-center gap-2 rounded-lg border border-border bg-card px-3 py-1.5"
