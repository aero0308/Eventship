   121→function teamTone(name: string): string {
   122→  let hash = 0
   123→  for (let i = 0; i < name.length; i += 1) hash = (hash * 31 + name.charCodeAt(i)) % 997
   124→  return TEAM_TONES[hash % TEAM_TONES.length]
   125→}

   127→/** Completion pill tone: emerald when mostly done, amber when started, neutral at 0. */

   136→interface RingEntry {
   137→  key: string
   138→  name: string
   139→  value: number

   142→/**
   143→ * Concentric "activity rings" for the priority breakdown: one ring per
   144→ * priority (HIGH outer → LOW inner), each filled by its share of all tasks,
   145→ * with the total count in the middle. Animated on mount, respects colors
   146→ * from PRIORITY_CHART_COLORS.
   147→ */
   148→function PriorityRings({ data }: { data: RingEntry[] }) {
   149→  const total = data.reduce((sum, entry) => sum + entry.value, 0)
   150→  const size = 208
   151→  const stroke = 15
   152→  const ringGap = 9
   153→  const scheme = useChartScheme()
   154→  const colors = PRIORITY_CHART_COLORS[scheme]
   155→
   156→  const rings = data.map((entry, index) => {
   157→    const radius = size / 2 - stroke / 2 - index * (stroke + ringGap)
   158→    const circumference = 2 * Math.PI * radius
   159→    return {
   160→      ...entry,
   161→      radius,
   162→      circumference,
   163→      share: total > 0 ? entry.value / total : 0,
   164→    }
   165→  })
   166→
   167→  return (
   168→    <div className="relative shrink-0" style={{ width: size, height: size }}>
   169→      <svg
   170→        viewBox={`0 0 ${size} ${size}`}
   171→        className="h-full w-full -rotate-90"
   172→        role="img"
   173→        aria-label={`Priority breakdown of ${total} tasks: ${data
   174→          .map((entry) => `${entry.value} ${entry.name.toLowerCase()}`)
   175→          .join(', ')}`}
   176→      >
   177→        {rings.map((ring) => (
   178→          <g key={ring.key}>
   179→            {/* Track */}
   180→            <circle cx={size / 2} cy={size / 2} r={ring.radius} fill="none" stroke="var(--muted)" strokeWidth={stroke} />
   181→            {/* Value arc — starts at 12 o'clock (svg rotated -90°) */}
   182→            <motion.circle
   183→              cx={size / 2}
   184→              cy={size / 2}
   185→              r={ring.radius}
   186→              fill="none"
   187→              stroke={colors[ring.key] ?? '#a8a29e'}
   188→              strokeWidth={stroke}
   189→              strokeLinecap="round"
   190→              strokeDasharray={ring.circumference}
   191→              initial={{ strokeDashoffset: ring.circumference }}
   192→              animate={{ strokeDashoffset: ring.circumference * (1 - ring.share) }}
   193→              transition={{ duration: 0.9, ease: 'easeOut', delay: 0.15 }}
   194→            />
   195→          </g>
   196→        ))}
   197→      </svg>
   198→      <div className="absolute inset-0 flex flex-col items-center justify-center" aria-hidden="true">
   199→        <span className="text-[2rem] font-bold leading-none tabular-nums text-foreground">{total}</span>
   200→        <span className="mt-1 text-[11px] font-medium uppercase tracking-wider text-muted-foreground">tasks</span>
   201→      </div>
   202→    </div>
   203→  )
   204→}

   314→function focusDueLabel(dueDate: string | null): string {
   315→  if (!dueDate) return 'No due date'
   316→  const diff = differenceInCalendarDays(new Date(dueDate), new Date())
   317→  if (diff === 0) return 'Due today'
   318→  if (diff === 1) return 'Due tomorrow'
   319→  if (diff < 0) return `Overdue by ${Math.abs(diff)}d`
   320→  return `Due ${format(new Date(dueDate), 'EEE, MMM d')}`
   321→}
