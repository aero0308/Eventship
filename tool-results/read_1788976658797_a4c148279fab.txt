   128→function byDue(a: string | null, b: string | null): number {
   129→  if (!a && !b) return 0
   130→  if (!a) return 1 // no due date sinks to the bottom
   131→  if (!b) return -1
   132→  return new Date(a).getTime() - new Date(b).getTime()
   133→}

   150→interface TimelineViewProps {
   151→  event: EventDTO
   152→  tasks: TaskDTO[]
   153→  expandedTaskId: string | null
   154→  onToggleExpand: (taskId: string) => void
   155→  onStatusChange: (task: TaskDTO, next: TaskStatus) => void

   159→/**
   160→ * Gantt-lite: one row per task; the bar runs from the event start to the task's
   161→ * due date (tasks without a due date get a dashed full-width "unslotted" bar).
   162→ * Shares the status filter/search results with the list view (passed via tasks).
   163→ */
   164→function TimelineView({ event, tasks, expandedTaskId, onToggleExpand, onStatusChange, rowUpdatingId }: TimelineViewProps) {
   165→  const start = new Date(event.startDate).getTime()
   166→  const end = Math.max(new Date(event.endDate).getTime(), start + DAY_MS)
   167→  const span = end - start
   168→  const pct = (t: number) => Math.min(100, Math.max(0, ((t - start) / span) * 100))
   169→
   170→  const now = Date.now()
   171→  const todayPct = now >= start && now <= end ? pct(now) : null
   172→
   173→  const ordered = useMemo(
   174→    () => [...tasks].sort((a, b) => byDue(a.dueDate, b.dueDate)),
   175→    [tasks]
   176→  )
   177→
   178→  const ticks = useMemo(() => {
   179→    // Adaptive tick count: fewer for short spans, more for long ones.
   180→    const dayCount = span / DAY_MS
   181→    const fractions =
   182→      dayCount <= 2 ? [0, 1] : dayCount <= 7 ? [0, 0.5, 1] : dayCount <= 45 ? [0, 0.25, 0.5, 0.75, 1] : [0, 1 / 6, 2 / 6, 3 / 6, 4 / 6, 5 / 6, 1]
   183→    const out: { pct: number; label: string }[] = []
   184→    for (const f of fractions) {
   185→      const date = new Date(start + f * span)
   186→      const label = format(date, 'MMM d')
   187→      // Skip a tick whose label collides with the previous one (short spans).
   188→      if (out.length > 0 && out[out.length - 1].label === label) continue
   189→      out.push({ pct: f * 100, label })
   190→    }
   191→    // Last tick colliding with the previous label → drop the earlier one.
   192→    if (out.length >= 2 && out[out.length - 1].label === out[out.length - 2].label) {
   193→      out.splice(out.length - 2, 1)
   194→    }
   195→    return out
   196→  }, [start, span])
   197→
   198→  return (
   199→    <div>
   200→      {/* Date axis */}
   201→      <div className="grid grid-cols-[8.5rem_1fr] gap-x-3 border-b border-border/60 px-4 pb-2 sm:grid-cols-[13rem_1fr]">
   202→        <div />
   203→        <div className="relative h-5" aria-hidden="true">
   204→          {ticks.map((tick) => (
   205→            <span
   206→              key={tick.pct}
   207→              className="absolute top-0 -translate-x-1/2 whitespace-nowrap text-[10px] font-medium text-muted-foreground/70"
   208→              style={{ left: `${tick.pct}%` }}
   209→            >
   210→              {tick.label}
   211→            </span>
   212→          ))}
   213→          {todayPct !== null ? (
   214→            <span
   215→              className="absolute -bottom-2 -translate-x-1/2 whitespace-nowrap rounded-full bg-amber-100 px-1.5 text-[9px] font-bold text-amber-800 ring-1 ring-amber-200 dark:bg-amber-500/20 dark:text-amber-300 dark:ring-amber-500/30"
   216→              style={{ left: `${todayPct}%` }}
   217→            >
   218→              Today
   219→            </span>
   220→          ) : null}
   221→        </div>
   222→      </div>
   223→
   224→      {/* Rows */}
   225→      <ul className="divide-y divide-border/40">
   226→        {ordered.map((task) => {
   227→          const due = task.dueDate ? new Date(task.dueDate) : null
   228→          const duePct = due ? pct(due.getTime()) : null
   229→          const overdue = due !== null && due.getTime() < Date.now() && task.status !== 'COMPLETED'
   230→          const expanded = expandedTaskId === task.id
   231→          return (
   232→            <li key={task.id}>
   233→              <button
   234→                type="button"
   235→                onClick={() => onToggleExpand(task.id)}
   236→                aria-expanded={expanded}
   237→                className={cn(
   238→                  'grid w-full grid-cols-[8.5rem_1fr] items-center gap-x-3 px-4 py-2.5 text-left transition-colors hover:bg-accent/40 sm:grid-cols-[13rem_1fr]',
   239→                  expanded && 'bg-accent/50'
   240→                )}
   241→              >
   242→                {/* Label */}
   243→                <span className="flex min-w-0 items-center gap-2">
   244→                  <Avatar className="h-5 w-5 shrink-0">
   245→                    <AvatarFallback className={cn('text-[8px] font-semibold', task.assignee ? 'bg-emerald-600 text-white' : 'bg-muted text-muted-foreground')}>
   246→                      {task.assignee ? initialsOf(task.assignee.fullName) : '—'}
   247→                    </AvatarFallback>
   248→                  </Avatar>
   249→                  <span className="min-w-0">
   250→                    <span className={cn('block truncate text-xs font-semibold text-foreground', task.status === 'COMPLETED' && 'text-muted-foreground line-through')}>
   251→                      {task.title}
   252→                    </span>
   253→                    <span className={cn('block truncate text-[10px]', overdue ? 'font-semibold text-red-600 dark:text-red-400' : 'text-muted-foreground/70')}>
   254→                      {due ? (overdue ? `Overdue · ${format(due, 'MMM d')}` : `Due ${format(due, 'MMM d')}`) : 'No due date'}
   255→                    </span>
   256→                  </span>
   257→                </span>
   258→
   259→                {/* Track */}
   260→                <span className="relative block h-6 rounded-md bg-muted/40 ring-1 ring-inset ring-border/50">
   261→                  {todayPct !== null ? (
   262→                    <span
   263→                      className="absolute inset-y-0 z-10 w-px border-l border-dashed border-amber-500/80"
   264→                      style={{ left: `${todayPct}%` }}
   265→                      aria-hidden="true"
   266→                    />
   267→                  ) : null}
   268→                  {duePct !== null ? (
   269→                    <motion.span
   270→                      initial={{ width: 0 }}
   271→                      animate={{ width: `${Math.max(duePct, 2)}%` }}
   272→                      transition={{ duration: 0.45, ease: 'easeOut' }}
   273→                      className={cn(
   274→                        'absolute inset-y-[5px] left-0 rounded-r-full rounded-l-sm bg-gradient-to-r shadow-sm transition-[filter] hover:brightness-105',
   275→                        TIMELINE_BAR[task.status] ?? TIMELINE_BAR.NOT_STARTED
   276→                      )}
   277→                      title={`${task.title} — due ${due ? format(due, 'MMM d, yyyy') : '—'} (${TASK_STATUS_LABELS[task.status] ?? task.status})`}
   278→                    />
   279→                  ) : (
   280→                    <span
   281→                      className="absolute inset-y-[6px] left-1 right-1 rounded-full border border-dashed border-stone-300 bg-transparent dark:border-stone-600"
   282→                      title={`${task.title} — no due date`}
   283→                    />
   284→                  )}
   285→                  {task.status === 'COMPLETED' && duePct !== null ? (
   286→                    <CheckCircle2
   287→                      className="absolute top-1/2 h-3.5 w-3.5 -translate-x-1/2 -translate-y-1/2 text-white drop-shadow"
   288→                      style={{ left: `${Math.max(duePct, 2)}%` }}
   289→                      aria-hidden="true"
   290→                    />
   291→                  ) : null}
   292→                </span>
   293→              </button>
   294→
   295→              {/* Expanded quick actions */}
   296→              {expanded ? (
   297→                <motion.div
   298→                  initial={{ opacity: 0, height: 0 }}
   299→                  animate={{ opacity: 1, height: 'auto' }}
   300→                  transition={{ duration: 0.2 }}
   301→                  className="overflow-hidden bg-accent/30"
   302→                >
   303→                  <div className="flex flex-wrap items-center gap-x-4 gap-y-2 px-4 py-3 sm:pl-[calc(8.5rem+1.75rem)] md:pl-[calc(13rem+1.75rem)]">
   304→                    <span className="flex items-center gap-1.5 text-xs text-muted-foreground">
   305→                      <span className={cn('rounded-full border px-2 py-0.5 text-[10px] font-semibold', PRIORITY_CLASSES[task.priority])}>
   306→                        {PRIORITY_LABELS[task.priority] ?? task.priority}
   307→                      </span>
   308→                      {task.assignee ? <span>· {task.assignee.fullName}</span> : <span>· Unassigned</span>}
   309→                      {(task.commentCount ?? 0) > 0 ? <span>· {task.commentCount} 💬</span> : null}
   310→                      {rowUpdatingId === task.id ? <Loader2 className="h-3.5 w-3.5 animate-spin" aria-hidden="true" /> : null}
   311→                    </span>
   312→                    <Select value={task.status} onValueChange={(value) => onStatusChange(task, value as TaskStatus)}>
   313→                      <SelectTrigger className="h-8 w-36 text-xs" aria-label={`Change status for ${task.title}`}>
   314→                        <SelectValue />
   315→                      </SelectTrigger>
   316→                      <SelectContent>
   317→                        {TASK_STATUSES.map((status) => (
   318→                          <SelectItem key={status} value={status}>
   319→                            {TASK_STATUS_LABELS[status]}
   320→                          </SelectItem>
   321→                        ))}
   322→                      </SelectContent>
   323→                    </Select>
   324→                    <button
   325→                      type="button"
   326→                      onClick={() => navigate(`${ROUTES.TASKS}?event=${event.id}`)}
   327→                      className="inline-flex h-8 items-center gap-1.5 rounded-md px-2 text-xs font-medium text-emerald-700 hover:bg-accent hover:text-emerald-800 dark:text-emerald-400 dark:hover:text-emerald-300"
   328→                    >
   329→                      <ExternalLink className="h-3.5 w-3.5" aria-hidden="true" />
   330→                      Open on board
   331→                    </button>
   332→                  </div>
   333→                </motion.div>
   334→              ) : null}
   335→            </li>
   336→          )
   337→        })}
   338→      </ul>
   339→
   340→      {/* Legend */}
   341→      <div className="flex flex-wrap items-center gap-x-4 gap-y-1.5 border-t border-border/60 px-4 py-2.5" aria-label="Timeline legend">
   342→        {(['NOT_STARTED', 'IN_PROGRESS', 'BLOCKED', 'COMPLETED'] as TaskStatus[]).map((status) => (
   343→          <span key={status} className="inline-flex items-center gap-1.5 text-[10px] font-medium text-muted-foreground">
   344→            <span className={cn('h-2.5 w-6 rounded-full bg-gradient-to-r', TIMELINE_BAR[status])} aria-hidden="true" />
   345→            {TASK_STATUS_LABELS[status]}
   346→          </span>
   347→        ))}
   348→        <span className="inline-flex items-center gap-1.5 text-[10px] font-medium text-muted-foreground">
   349→          <span className="h-3 w-0 border-l border-dashed border-amber-500" aria-hidden="true" />
   350→          Today
   351→        </span>
   352→        <span className="inline-flex items-center gap-1.5 text-[10px] font-medium text-muted-foreground">
   353→          <span className="h-2.5 w-6 rounded-full border border-dashed border-stone-300 dark:border-stone-600" aria-hidden="true" />
   354→          No due date
   355→        </span>
   356→      </div>
   357→    </div>
   358→  )
