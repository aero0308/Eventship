   113→   113→function dueChip(task: TaskDTO): { label: string; classes: string } | null {
   114→   114→  if (!task.dueDate) return null
   115→   115→  const due = new Date(task.dueDate)
   116→   116→  const days = differenceInCalendarDays(due, new Date())
   117→   117→  const done = task.status === 'COMPLETED'
   118→   118→  if (done) {
   119→   119→    return { label: `Due ${format(due, 'MMM d')}`, classes: 'border-border bg-muted/50 text-muted-foreground/70' }
   120→   120→  }
   121→   121→  if (days < 0) return { label: `${Math.abs(days)}d overdue`, classes: 'border-red-200 bg-red-50 text-red-700' }
   122→   122→  if (days === 0) return { label: 'Due today', classes: 'border-amber-200 bg-amber-50 text-amber-800' }
   123→   123→  if (days < 3) return { label: `${days}d left`, classes: 'border-amber-200 bg-amber-50 text-amber-800' }
   124→   124→  return { label: `Due ${format(due, 'MMM d')}`, classes: 'border-border bg-muted/50 text-muted-foreground' }
   125→   125→}
