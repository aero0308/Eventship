   156→import { useState, useRef, useEffect } from 'react'
   157→
   158→interface DropdownItem {
   159→    label: string
   160→    onClick: () => void
   161→    icon?: React.ReactNode
   162→    variant?: 'default' | 'danger'
   163→    disabled?: boolean
   164→}
   165→
   166→interface DropdownProps {
   167→    trigger: React.ReactNode
   168→    items: DropdownItem[]
   169→    align?: 'left' | 'right'
   170→}
   171→
   172→export const Dropdown = ({ trigger, items, align = 'right' }: DropdownProps) => {
   173→    const [isOpen, setIsOpen] = useState(false)
   174→    const dropdownRef = useRef<HTMLDivElement>(null)
   175→    
   176→    useEffect(() => {
   177→        const handleClickOutside = (e: MouseEvent) => {
   178→            if (dropdownRef.current && !dropdownRef.current.contains(e.target as Node)) {
   179→                setIsOpen(false)
   180→            }
   181→        }
   182→        document.addEventListener('mousedown', handleClickOutside)
   183→        return () => document.removeEventListener('mousedown', handleClickOutside)
   184→    }, [])
   185→    
   186→    return (
   187→        <div className="relative" ref={dropdownRef}>
   188→            <div onClick={() => setIsOpen(!isOpen)}>{trigger}</div>
   189→            {isOpen && (
   190→                <div className={`absolute z-50 mt-2 w-48 bg-white rounded-lg shadow-lg border border-gray-200 py-1 ${align === 'right' ? 'right-0' : 'left-0'}`}>
   191→                    {items.map((item, idx) => (
   192→                        <button
   193→                            key={idx}
   194→                            onClick={() => { item.onClick(); setIsOpen(false) }}
   195→                            disabled={item.disabled}
   196→                            className={`w-full text-left px-4 py-2 text-sm hover:bg-gray-50 disabled:opacity-50 ${item.variant === 'danger' ? 'text-red-600' : 'text-gray-700'}`}
   197→                        >
   198→                            {item.icon && <span className="mr-2">{item.icon}</span>}
   199→                            {item.label}
   200→                        </button>
   201→                    ))}
   202→                </div>
   203→            )}
   204→        </div>
   205→    )

   844→def add_dependency(
   845→    db: Session,
   846→    task_id: UUID,
   847→    depends_on_task_id: UUID
   848→) -> TaskDependency:
   849→    """Add dependency (task depends on another task)"""
   850→    if task_id == depends_on_task_id:
   851→        raise HTTPException(status_code=400, detail="Task cannot depend on itself")
   852→    
   853→    # Check both tasks exist
   854→    task = get_task(db, task_id)
   855→    depends_on = get_task(db, depends_on_task_id)
   856→    if not task or not depends_on:
   857→        raise HTTPException(status_code=404, detail="Task not found")
   858→    
   859→    # Check for circular dependency
   860→    if _has_circular_dependency(db, task_id, depends_on_task_id):
   861→        raise HTTPException(status_code=400, detail="Circular dependency detected")
   862→    
   863→    # Check if dependency already exists
   864→    existing = db.query(TaskDependency).filter(
   865→        TaskDependency.task_id == task_id,
   866→        TaskDependency.depends_on_task_id == depends_on_task_id
   867→    ).first()
   868→    if existing:
   869→        raise HTTPException(status_code=400, detail="Dependency already exists")
   870→    
   871→    dependency = TaskDependency(
   872→        task_id=task_id,
   873→        depends_on_task_id=depends_on_task_id
   874→    )
   875→    db.add(dependency)
   876→    db.commit()
   877→    db.refresh(dependency)
   878→    return dependency

   880→def _has_circular_dependency(db: Session, task_id: UUID, depends_on_id: UUID) -> bool:
   881→    """Check if adding this dependency creates a cycle"""
   882→    visited = set()
   883→    to_visit = [depends_on_id]
   884→    
   885→    while to_visit:
   886→        current = to_visit.pop()
   887→        if current == task_id:
   888→            return True
   889→        if current in visited:
   890→            continue
   891→        visited.add(current)
   892→        
   893→        deps = db.query(TaskDependency).filter(
   894→            TaskDependency.task_id == current
   895→        ).all()
   896→        to_visit.extend([d.depends_on_task_id for d in deps])
   897→    
   898→    return False

  1799→        } else {
  1800→            // If dropped on another task, get that task's status
  1801→            const overTask = tasks.find(t => t.id === over.id)
  1802→            if (!overTask) return
  1803→            newStatus = overTask.status

  1816→        // If moving to COMPLETED, check for incomplete dependencies
