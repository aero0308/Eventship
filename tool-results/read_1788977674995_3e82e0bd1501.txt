    50→    50→function monthKeyOf(date: Date): string {
    51→    51→  return `${date.getFullYear()}-${String(date.getMonth() + 1).padStart(2, '0')}`
    52→    52→}
    53→    53→
    54→    54→function dayKeyOf(date: Date): string {
    55→    55→  return `${date.getFullYear()}-${String(date.getMonth() + 1).padStart(2, '0')}-${String(date.getDate()).padStart(2, '0')}`
    56→    56→}
    57→    57→
    58→    58→/** Monday-start offset of the 1st of the given month. */
    59→    59→function firstWeekdayOffset(year: number, monthIndex: number): number {
    60→    60→  return (new Date(year, monthIndex, 1).getDay() + 6) % 7
    61→    61→}
    62→    62→
    63→    63→function shiftMonth(monthKey: string, delta: number): string {
    64→    64→  const [y, m] = monthKey.split('-').map(Number)
    65→    65→  const d = new Date(y, m - 1 + delta, 1)
    66→    66→  return monthKeyOf(d)
    67→    67→}

   398→   398→                      const weekend = day.getDay() === 0 || day.getDay() === 6
