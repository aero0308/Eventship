interface ProgressBarProps {
    value: number // 0-100
    showLabel?: boolean
    size?: 'sm' | 'md' | 'lg'
    variant?: 'default' | 'success' | 'warning' | 'danger'
}

export const ProgressBar = ({ 
    value, 
    showLabel = true, 
    size = 'md',
    variant = 'default' 
}: ProgressBarProps) => {
    const heights = { sm: 'h-1', md: 'h-2', lg: 'h-3' }
    const colors = {
        default: 'bg-blue-600',
        success: 'bg-green-600',
        warning: 'bg-yellow-600',
        danger: 'bg-red-600'
    }
    
    return (
        <div>
            <div className={`w-full bg-gray-200 rounded-full ${heights[size]}`}>
                <div
                    className={`${colors[variant]} ${heights[size]} rounded-full transition-all`}
                    style={{ width: `${Math.min(100, Math.max(0, value))}%` }}
                />
            </div>
            {showLabel && (
                <p className="text-xs text-gray-500 mt-1">{value}% complete</p>
            )}
        </div>
    )
}

from pydantic import BaseModel, Field, validator
from typing import Optional
from datetime import datetime
from uuid import UUID
from enum import Enum

class EventStatus(str, Enum):
    DRAFT = "DRAFT"
    PLANNING = "PLANNING"
    IN_PROGRESS = "IN_PROGRESS"
    COMPLETED = "COMPLETED"
    CANCELLED = "CANCELLED"

class EventBase(BaseModel):
    name: str = Field(..., min_length=3, max_length=200)
    description: Optional[str] = Field(None, max_length=1000)
    start_date: datetime
    end_date: datetime
    status: EventStatus = EventStatus.DRAFT
    team_id: UUID

    @validator('end_date')
    def validate_dates(cls, v, values):
        if 'start_date' in values and v < values['start_date']:
            raise ValueError('end_date must be after start_date')
        return v

class EventCreate(EventBase):
    pass

class EventUpdate(BaseModel):
    name: Optional[str] = Field(None, min_length=3, max_length=200)

    start_date: Optional[datetime] = None
    end_date: Optional[datetime] = None

import { create } from 'zustand'
import { eventService } from '../services/eventService'
import { 
    Event, EventCreate, EventUpdate, 
    EventProgress, EventFilters 
} from '../types/event'

interface EventState {
    // State
    events: Event[]
    currentEvent: Event | null
    eventProgress: EventProgress | null
    eventTasks: any[]
    upcomingEvents: Event[]
    filters: EventFilters
    isLoading: boolean
    error: string | null
    
    // Actions
    fetchEvents: (params?: EventFilters) => Promise<void>
    fetchEvent: (id: string) => Promise<void>
    createEvent: (data: EventCreate) => Promise<Event>
    updateEvent: (id: string, data: EventUpdate) => Promise<Event>
    deleteEvent: (id: string) => Promise<void>
    updateEventStatus: (id: string, status: any) => Promise<void>
    fetchEventProgress: (id: string) => Promise<void>
    fetchEventTasks: (id: string, params?: any) => Promise<void>
    fetchUpcomingEvents: (days?: number) => Promise<void>
    setFilters: (filters: EventFilters) => void
    clearFilters: () => void
    clearCurrentEvent: () => void
    clearError: () => void
}

export const useEventStore = create<EventState>((set, get) => ({
    events: [],
    currentEvent: null,
    eventProgress: null,
    eventTasks: [],
    upcomingEvents: [],
    filters: {},
    isLoading: false,
    error: null,
    
    fetchEvents: async (params) => {
        set({ isLoading: true, error: null })
        try {
            const events = await eventService.getEvents(params)
            set({ events, isLoading: false })
        } catch (error: any) {
            set({ 
                error: error.response?.data?.detail || 'Failed to fetch events', 
                isLoading: false 
            })
        }
    },
    
    fetchEvent: async (id) => {
        set({ isLoading: true, error: null })
        try {
            const event = await eventService.getEvent(id)
            set({ currentEvent: event, isLoading: false })
        } catch (error: any) {
            set({ 
                error: error.response?.data?.detail || 'Failed to fetch event', 
                isLoading: false 
            })
        }
    },
    
    createEvent: async (data) => {
        set({ isLoading: true, error: null })
        try {
            const event = await eventService.createEvent(data)
            set(state => ({ 
                events: [event, ...state.events], 
                isLoading: false 
            }))
            return event
        } catch (error: any) {
            set({ 
                error: error.response?.data?.detail || 'Failed to create event', 
                isLoading: false 
            })
            throw error
        }
    },
    
    updateEvent: async (id, data) => {
        set({ isLoading: true, error: null })
        try {
            const updated = await eventService.updateEvent(id, data)
            set(state => ({
                events: state.events.map(e => e.id === id ? updated : e),
                currentEvent: state.currentEvent?.id === id ? updated : state.currentEvent,
                isLoading: false
            }))
            return updated
        } catch (error: any) {
            set({ 
                error: error.response?.data?.detail || 'Failed to update event', 
                isLoading: false 
            })
            throw error
        }
    },
    
    deleteEvent: async (id) => {
        set({ isLoading: true, error: null })
        try {
            await eventService.deleteEvent(id)
            set(state => ({
                events: state.events.filter(e => e.id !== id),
                currentEvent: state.currentEvent?.id === id ? null : state.currentEvent,
                isLoading: false
            }))
        } catch (error: any) {
            set({ 
                error: error.response?.data?.detail || 'Failed to delete event', 
                isLoading: false 
            })
            throw error
        }
    },
    
    updateEventStatus: async (id, status) => {
        try {
            const updated = await eventService.updateEventStatus(id, status)
            set(state => ({
                events: state.events.map(e => e.id === id ? updated : e),
                currentEvent: state.currentEvent?.id === id ? updated : state.currentEvent
            }))
        } catch (error: any) {
            set({ error: error.response?.data?.detail || 'Failed to update status' })
            throw error
        }
    },
    
    fetchEventProgress: async (id) => {
        try {
            const progress = await eventService.getEventProgress(id)
            set({ eventProgress: progress })
        } catch (error: any) {
            set({ error: error.response?.data?.detail || 'Failed to fetch progress' })
        }
    },
    
    fetchEventTasks: async (id, params) => {
        try {
            const tasks = await eventService.getEventTasks(id, params)
            set({ eventTasks: tasks })
        } catch (error: any) {
            set({ error: error.response?.data?.detail || 'Failed to fetch tasks' })
        }
    },
    
    fetchUpcomingEvents: async (days = 30) => {
        try {
            const events = await eventService.getUpcomingEvents(days)
            set({ upcomingEvents: events })
        } catch (error: any) {
            set({ error: error.response?.data?.detail || 'Failed to fetch upcoming' })
        }
    },
    
    setFilters: (filters) => set({ filters }),
    clearFilters: () => set({ filters: {} }),
    clearCurrentEvent: () => set({ currentEvent: null, eventProgress: null, eventTasks: [] }),
    clearError: () => set({ error: null })
}))

import { useEffect } from 'react'
import { useEventStore } from '../stores/eventStore'
import { EventFilters } from '../types/event'

export const useEvents = (filters?: EventFilters) => {
    const { events, isLoading, error, fetchEvents } = useEventStore()
    
    useEffect(() => {
        fetchEvents(filters)
    }, [filters])
    
    return { events, isLoading, error, refetch: () => fetchEvents(filters) }
}

export const useEvent = (id: string) => {
    const { currentEvent, isLoading, error, fetchEvent } = useEventStore()
    
    useEffect(() => {
        if (id) fetchEvent(id)
    }, [id])
    
    return { event: currentEvent, isLoading, error, refetch: () => fetchEvent(id) }
}

export const useUpcomingEvents = (days = 30) => {
    const { upcomingEvents, isLoading, fetchUpcomingEvents } = useEventStore()
    
    useEffect(() => {
        fetchUpcomingEvents(days)
    }, [days])
    
    return { events: upcomingEvents, isLoading }
}

const eventSchema = z.object({
    name: z.string().min(3, 'Name must be at least 3 characters').max(200),
    description: z.string().max(1000).optional(),
    start_date: z.string().min(1, 'Start date is required'),
    end_date: z.string().min(1, 'End date is required'),
    status: z.enum(['DRAFT', 'PLANNING', 'IN_PROGRESS', 'COMPLETED', 'CANCELLED']),
    team_id: z.string().min(1, 'Team is required')
}).refine((data) => {
    if (data.start_date && data.end_date) {
        return new Date(data.end_date) > new Date(data.start_date)
    }
    return true
}, {
    message: 'End date must be after start date',
    path: ['end_date']
})

interface EventFormProps {

    isLoading?: boolean

## 🚨 Common Issues & Fixes

### Issue 1: "End date must be after start date"
**Cause:** Invalid date order
**Fix:** Ensure end date > start date

### Issue 2: 403 Forbidden on Create
**Cause:** Not EVENT_MANAGER or TEAM_LEADER
**Fix:** Login with proper role

### Issue 3: Team dropdown empty
**Cause:** No teams created yet
**Fix:** Create a team in Phase 3 first

### Issue 4: 400 on Delete
**Cause:** Event has tasks
**Fix:** Delete tasks first (or wait for Phase 5 cascade)

### Issue 5: Dates showing wrong
**Cause:** Timezone issues
**Fix:** Use ISO format in backend, format in frontend
