from sqlalchemy.orm import Session
from sqlalchemy import func, and_, or_
from typing import List, Optional, Dict
from uuid import UUID
from datetime import datetime, timedelta

from app.models.task import Task, TaskStatus, TaskPriority
from app.models.event import Event, EventStatus
from app.models.team import Team
from app.models.user import User, UserRole
from app.models.activity import ActivityLog
from app.models.comment import TaskComment


def get_dashboard_stats(db: Session, user: Optional[User] = None) -> Dict:
    """
    Get overall dashboard statistics.
    
    Authorization:
    - EVENT_MANAGER: sees all data
    - TEAM_LEADER: sees team data
    - EMPLOYEE: sees only their tasks
    """
    # Base queries with role-based filtering
    if user and user.role == UserRole.EMPLOYEE:
        # Employee: only their tasks
        task_query = db.query(Task).filter(Task.assigned_to == user.id)
        event_query = db.query(Event).join(Team).filter(Team.id == user.team_id)
        team_query = db.query(Team).filter(Team.id == user.team_id)
        user_query = db.query(User).filter(User.team_id == user.team_id)
    elif user and user.role == UserRole.TEAM_LEADER:
        # Team leader: team's data
        task_query = db.query(Task).join(Event).filter(Event.team_id == user.team_id)
        event_query = db.query(Event).filter(Event.team_id == user.team_id)
        team_query = db.query(Team).filter(Team.id == user.team_id)
        user_query = db.query(User).filter(User.team_id == user.team_id)
    else:
        # Manager: all data
        task_query = db.query(Task)
        event_query = db.query(Event)
        team_query = db.query(Team)
        user_query = db.query(User)
    
    # Count stats
    total_tasks = task_query.count()
    completed_tasks = task_query.filter(Task.status == TaskStatus.COMPLETED).count()
    in_progress_tasks = task_query.filter(Task.status == TaskStatus.IN_PROGRESS).count()
    blocked_tasks = task_query.filter(Task.status == TaskStatus.BLOCKED).count()
    not_started_tasks = task_query.filter(Task.status == TaskStatus.NOT_STARTED).count()
    overdue_tasks = task_query.filter(
        Task.due_date < datetime.utcnow(),
        Task.status != TaskStatus.COMPLETED
    ).count()
    
    total_events = event_query.count()
    active_events = event_query.filter(
        Event.status.in_([EventStatus.PLANNING, EventStatus.IN_PROGRESS])
    ).count()
    
    total_teams = team_query.count()
    total_users = user_query.count()
    
    completion_rate = (completed_tasks / total_tasks * 100) if total_tasks > 0 else 0
    
    return {
        "total_events": total_events,
        "active_events": active_events,
        "total_tasks": total_tasks,
        "completed_tasks": completed_tasks,
        "in_progress_tasks": in_progress_tasks,
        "blocked_tasks": blocked_tasks,
        "not_started_tasks": not_started_tasks,
        "overdue_tasks": overdue_tasks,
        "completion_rate": round(completion_rate, 2),
        "total_teams": total_teams,
        "total_users": total_users
    }


def get_task_distribution(db: Session, user: Optional[User] = None) -> Dict:
    """Get task count by status for chart"""
    query = db.query(Task)
    
    # Apply role filter
    if user:
        if user.role == UserRole.EMPLOYEE:
            query = query.filter(Task.assigned_to == user.id)
        elif user.role == UserRole.TEAM_LEADER:
            query = query.join(Event).filter(Event.team_id == user.team_id)
    
    return {
        "not_started": query.filter(Task.status == TaskStatus.NOT_STARTED).count(),
        "in_progress": query.filter(Task.status == TaskStatus.IN_PROGRESS).count(),
        "blocked": query.filter(Task.status == TaskStatus.BLOCKED).count(),
        "completed": query.filter(Task.status == TaskStatus.COMPLETED).count()
    }


def get_priority_distribution(db: Session, user: Optional[User] = None) -> Dict:
    """Get task count by priority for chart"""
    query = db.query(Task)
    
    if user:
        if user.role == UserRole.EMPLOYEE:
            query = query.filter(Task.assigned_to == user.id)
        elif user.role == UserRole.TEAM_LEADER:
            query = query.join(Event).filter(Event.team_id == user.team_id)
    
    return {
        "high": query.filter(Task.priority == TaskPriority.HIGH).count(),
        "medium": query.filter(Task.priority == TaskPriority.MEDIUM).count(),
        "low": query.filter(Task.priority == TaskPriority.LOW).count()
    }


def get_upcoming_deadlines(
    db: Session,
    user: Optional[User] = None,
    days: int = 7,
    limit: int = 10
) -> List[Dict]:
    """Get tasks due in the next N days"""
    now = datetime.utcnow()
    future = now + timedelta(days=days)
    
    query = db.query(Task).filter(
        Task.due_date.between(now, future),
        Task.status != TaskStatus.COMPLETED
    )
    
    # Role-based filtering
    if user:
        if user.role == UserRole.EMPLOYEE:
            query = query.filter(Task.assigned_to == user.id)
        elif user.role == UserRole.TEAM_LEADER:
            query = query.join(Event).filter(Event.team_id == user.team_id)
    
    tasks = query.order_by(Task.due_date.asc()).limit(limit).all()
    
    return [{
        "id": t.id,
        "title": t.title,
        "due_date": t.due_date,
        "priority": t.priority.value,
        "status": t.status.value,
        "event_name": t.event.name if t.event else None,
        "assignee_name": t.assignee.full_name if t.assignee else None,
        "days_until_due": (t.due_date - now).days,
        "is_overdue": False
    } for t in tasks]


def get_blockers(
    db: Session,
    user: Optional[User] = None,
    limit: int = 10
) -> List[Dict]:
    """Get all blocked tasks with details"""
    query = db.query(Task).filter(Task.status == TaskStatus.BLOCKED)
    
    if user:
        if user.role == UserRole.EMPLOYEE:
            query = query.filter(Task.assigned_to == user.id)
        elif user.role == UserRole.TEAM_LEADER:
            query = query.join(Event).filter(Event.team_id == user.team_id)
    
    tasks = query.order_by(Task.updated_at.desc()).limit(limit).all()
    
    result = []
    for t in tasks:
        # Get latest comment
        latest_comment = db.query(TaskComment).filter(
            TaskComment.task_id == t.id
        ).order_by(TaskComment.created_at.desc()).first()
        
        # Calculate days blocked (since last status change)
        days_blocked = (datetime.utcnow() - t.updated_at).days if t.updated_at else 0
        
        result.append({
            "id": t.id,
            "title": t.title,
            "priority": t.priority.value,
            "event_name": t.event.name if t.event else None,
            "assignee_name": t.assignee.full_name if t.assignee else None,
            "due_date": t.due_date,
            "days_blocked": days_blocked,
            "latest_comment": latest_comment.content if latest_comment else None
        })
    
    return result


def get_recent_activity(
    db: Session,
    user: Optional[User] = None,
    limit: int = 20
) -> List[Dict]:
    """Get recent activity logs"""
    query = db.query(ActivityLog)
    
    # Role-based filtering
    if user and user.role == UserRole.EMPLOYEE:
        # Employee sees activity related to their tasks
        task_ids = db.query(Task.id).filter(Task.assigned_to == user.id).subquery()
        query = query.filter(
            or_(
                ActivityLog.user_id == user.id,
                ActivityLog.details['task_id'].astext.in_(
                    db.query(Task.id).filter(Task.assigned_to == user.id)
                )
            )
        )
    
    activities = query.order_by(ActivityLog.timestamp.desc()).limit(limit).all()
    
    return [{
        "id": a.id,
        "action": a.action,
        "details": a.details,
        "user_name": a.user.full_name if a.user else "Unknown",
        "user_avatar": None,
        "timestamp": a.timestamp,
        "related_task_id": a.details.get("task_id") if a.details else None,
        "related_event_id": a.details.get("event_id") if a.details else None
    } for a in activities]


def get_team_performance(
    db: Session,
    user: Optional[User] = None,
    limit: int = 10
) -> List[Dict]:
    """Get performance metrics per team (managers only)"""
    teams = db.query(Team).limit(limit).all()
    
    result = []
    for team in teams:
        total_tasks = db.query(Task).join(Event).filter(Event.team_id == team.id).count()
        completed_tasks = db.query(Task).join(Event).filter(
            Event.team_id == team.id,
            Task.status == TaskStatus.COMPLETED
        ).count()
        blocked_tasks = db.query(Task).join(Event).filter(
            Event.team_id == team.id,
            Task.status == TaskStatus.BLOCKED
        ).count()
        member_count = db.query(User).filter(User.team_id == team.id).count()
        
        completion_rate = (completed_tasks / total_tasks * 100) if total_tasks > 0 else 0
        
        result.append({
            "team_id": team.id,
            "team_name": team.name,
            "member_count": member_count,
            "total_tasks": total_tasks,
            "completed_tasks": completed_tasks,
            "blocked_tasks": blocked_tasks,
            "completion_rate": round(completion_rate, 2)
        })
    
    return result


def get_top_performers(
    db: Session,
    user: Optional[User] = None,
    limit: int = 5
) -> List[Dict]:
    """Get top performing employees (managers only)"""
    # Query users with their task counts
    users = db.query(User).filter(User.role == UserRole.EMPLOYEE)
    
    if user and user.role == UserRole.TEAM_LEADER:
        users = users.filter(User.team_id == user.team_id)
    
    result = []
    for u in users.limit(50).all():  # Limit initial query
        total = db.query(Task).filter(Task.assigned_to == u.id).count()
        completed = db.query(Task).filter(
            Task.assigned_to == u.id,
            Task.status == TaskStatus.COMPLETED
        ).count()
        in_progress = db.query(Task).filter(
            Task.assigned_to == u.id,
            Task.status == TaskStatus.IN_PROGRESS
        ).count()
        blocked = db.query(Task).filter(
            Task.assigned_to == u.id,
            Task.status == TaskStatus.BLOCKED
        ).count()
        
        if total > 0:
            completion_rate = (completed / total * 100)
            result.append({
                "user_id": u.id,
                "user_name": u.full_name,
                "user_avatar": None,
                "total_tasks": total,
                "completed_tasks": completed,
                "in_progress_tasks": in_progress,
                "blocked_tasks": blocked,
                "completion_rate": round(completion_rate, 2)
            })
    
    # Sort by completion rate descending
    result.sort(key=lambda x: x["completion_rate"], reverse=True)
    return result[:limit]


def get_progress_over_time(
    db: Session,
    user: Optional[User] = None,
    days: int = 30
) -> List[Dict]:
    """Get daily progress data for the last N days"""
    result = []
    now = datetime.utcnow()
    
    for i in range(days, -1, -1):
        date = now - timedelta(days=i)
        date_start = date.replace(hour=0, minute=0, second=0, microsecond=0)
        date_end = date_start + timedelta(days=1)
        
        query_created = db.query(Task).filter(
            Task.created_at.between(date_start, date_end)
        )
        query_completed = db.query(Task).filter(
            Task.status == TaskStatus.COMPLETED,
            Task.updated_at.between(date_start, date_end)
        )
        
        if user:
            if user.role == UserRole.EMPLOYEE:
                query_created = query_created.filter(Task.assigned_to == user.id)
                query_completed = query_completed.filter(Task.assigned_to == user.id)
            elif user.role == UserRole.TEAM_LEADER:
                query_created = query_created.join(Event).filter(Event.team_id == user.team_id)
                query_completed = query_completed.join(Event).filter(Event.team_id == user.team_id)
        
        result.append({
            "date": date_start.strftime("%Y-%m-%d"),
            "created": query_created.count(),
            "completed": query_completed.count(),
            "cumulative_completed": 0  # Will calculate below
        })
    
    # Calculate cumulative
    cumulative = 0
    for item in result:
        cumulative += item["completed"]
        item["cumulative_completed"] = cumulative
    
    return result

/**

 */
export const useAutoRefreshDashboard = (intervalMs: number = 30000) => {
    const refreshDashboard = useDashboardStore(state => state.refreshDashboard)
    const intervalRef = useRef<NodeJS.Timeout>()
    
    useEffect(() => {
        intervalRef.current = setInterval(() => {
            refreshDashboard()
        }, intervalMs)

                clearInterval(intervalRef.current)
            }
        }
    }, [intervalMs, refreshDashboard])
