import api from './api'
import { Team, TeamCreate, TeamUpdate, TeamMember, TeamStats } from '../types/team'

export const teamService = {
    // Get all teams (with optional filters)
    getTeams: async (params?: { search?: string; manager_id?: string }) => {
        const response = await api.get('/teams', { params })
        return response.data
    },
    
    // Get single team by ID
    getTeam: async (id: string) => {
        const response = await api.get(`/teams/${id}`)
        return response.data
    },
    
    // Create new team
    createTeam: async (data: TeamCreate) => {
        const response = await api.post('/teams', data)
        return response.data
    },
    
    // Update team
    updateTeam: async (id: string, data: TeamUpdate) => {
        const response = await api.put(`/teams/${id}`, data)
        return response.data
    },
    
    // Delete team
    deleteTeam: async (id: string) => {
        const response = await api.delete(`/teams/${id}`)
        return response.data
    },
    
    // Get team members
    getTeamMembers: async (teamId: string) => {
        const response = await api.get(`/teams/${teamId}/members`)
        return response.data
    },
    
    // Add member to team
    addMember: async (teamId: string, userId: string) => {
        const response = await api.post(`/teams/${teamId}/members`, { user_id: userId })
        return response.data
    },
    
    // Remove member from team
    removeMember: async (teamId: string, userId: string) => {
        const response = await api.delete(`/teams/${teamId}/members/${userId}`)
        return response.data
    },
    
    // Get team statistics
    getTeamStats: async (teamId: string) => {
        const response = await api.get(`/teams/${teamId}/stats`)
        return response.data
    }
}

import { create } from 'zustand'
import { teamService } from '../services/teamService'
import { Team, TeamCreate, TeamUpdate, TeamMember, TeamStats } from '../types/team'

interface TeamState {
    // State
    teams: Team[]
    currentTeam: Team | null
    teamMembers: TeamMember[]
    teamStats: TeamStats | null
    isLoading: boolean
    error: string | null
    
    // Actions
    fetchTeams: (params?: any) => Promise<void>
    fetchTeam: (id: string) => Promise<void>
    createTeam: (data: TeamCreate) => Promise<Team>
    updateTeam: (id: string, data: TeamUpdate) => Promise<Team>
    deleteTeam: (id: string) => Promise<void>
    fetchTeamMembers: (teamId: string) => Promise<void>
    addMember: (teamId: string, userId: string) => Promise<void>
    removeMember: (teamId: string, userId: string) => Promise<void>
    fetchTeamStats: (teamId: string) => Promise<void>
    clearError: () => void
    clearCurrentTeam: () => void
}

export const useTeamStore = create<TeamState>((set, get) => ({
    // Initial state
    teams: [],
    currentTeam: null,
    teamMembers: [],
    teamStats: null,
    isLoading: false,
    error: null,
    
    // Fetch all teams
    fetchTeams: async (params) => {
        set({ isLoading: true, error: null })
        try {
            const teams = await teamService.getTeams(params)
            set({ teams, isLoading: false })
        } catch (error: any) {
            set({ 
                error: error.response?.data?.detail || 'Failed to fetch teams', 
                isLoading: false 
            })
        }
    },
    
    // Fetch single team
    fetchTeam: async (id) => {
        set({ isLoading: true, error: null })
        try {
            const team = await teamService.getTeam(id)
            set({ currentTeam: team, isLoading: false })
        } catch (error: any) {
            set({ 
                error: error.response?.data?.detail || 'Failed to fetch team', 
                isLoading: false 
            })
        }
    },
    
    // Create team
    createTeam: async (data) => {
        set({ isLoading: true, error: null })
        try {
            const team = await teamService.createTeam(data)
            set(state => ({ 
                teams: [...state.teams, team], 
                isLoading: false 
            }))
            return team
        } catch (error: any) {
            set({ 
                error: error.response?.data?.detail || 'Failed to create team', 
                isLoading: false 
            })
            throw error
        }
    },
    
    // Update team
    updateTeam: async (id, data) => {
        set({ isLoading: true, error: null })
        try {
            const updatedTeam = await teamService.updateTeam(id, data)
            set(state => ({
                teams: state.teams.map(t => t.id === id ? updatedTeam : t),
                currentTeam: state.currentTeam?.id === id ? updatedTeam : state.currentTeam,
                isLoading: false
            }))
            return updatedTeam
        } catch (error: any) {
            set({ 
                error: error.response?.data?.detail || 'Failed to update team', 
                isLoading: false 
            })
            throw error
        }
    },
    
    // Delete team
    deleteTeam: async (id) => {
        set({ isLoading: true, error: null })
        try {
            await teamService.deleteTeam(id)
            set(state => ({
                teams: state.teams.filter(t => t.id !== id),
                currentTeam: state.currentTeam?.id === id ? null : state.currentTeam,
                isLoading: false
            }))
        } catch (error: any) {
            set({ 
                error: error.response?.data?.detail || 'Failed to delete team', 
                isLoading: false 
            })
            throw error
        }
    },
    
    // Fetch team members
    fetchTeamMembers: async (teamId) => {
        set({ isLoading: true, error: null })
        try {
            const members = await teamService.getTeamMembers(teamId)
            set({ teamMembers: members, isLoading: false })
        } catch (error: any) {
            set({ 
                error: error.response?.data?.detail || 'Failed to fetch team members', 
                isLoading: false 
            })
        }
    },
    
    // Add member to team
    addMember: async (teamId, userId) => {
        set({ isLoading: true, error: null })
        try {
            await teamService.addMember(teamId, userId)
            // Refresh members list
            const members = await teamService.getTeamMembers(teamId)
            set({ teamMembers: members, isLoading: false })
        } catch (error: any) {
            set({ 
                error: error.response?.data?.detail || 'Failed to add member', 
                isLoading: false 
            })
            throw error
        }
    },
    
    // Remove member from team
    removeMember: async (teamId, userId) => {
        set({ isLoading: true, error: null })
        try {
            await teamService.removeMember(teamId, userId)
            set(state => ({
                teamMembers: state.teamMembers.filter(m => m.id !== userId),
                isLoading: false
            }))
        } catch (error: any) {
            set({ 
                error: error.response?.data?.detail || 'Failed to remove member', 
                isLoading: false 
            })
            throw error
        }
    },
    
    // Fetch team stats
    fetchTeamStats: async (teamId) => {
        try {
            const stats = await teamService.getTeamStats(teamId)
            set({ teamStats: stats })
        } catch (error: any) {
            set({ error: error.response?.data?.detail || 'Failed to fetch stats' })
        }
    },
    
    // Clear error
    clearError: () => set({ error: null }),
    
    // Clear current team
    clearCurrentTeam: () => set({ currentTeam: null, teamMembers: [], teamStats: null })
}))

import { useEffect } from 'react'
import { useTeamStore } from '../stores/teamStore'

export const useTeams = (params?: any) => {
    const { teams, isLoading, error, fetchTeams } = useTeamStore()
    
    useEffect(() => {
        fetchTeams(params)
    }, [])
    
    return { teams, isLoading, error, refetch: () => fetchTeams(params) }
}

export const useTeam = (id: string) => {
    const { currentTeam, isLoading, error, fetchTeam } = useTeamStore()
    
    useEffect(() => {
        if (id) fetchTeam(id)
    }, [id])
    
    return { team: currentTeam, isLoading, error, refetch: () => fetchTeam(id) }
}

export const useTeamMembers = (teamId: string) => {
    const { teamMembers, isLoading, fetchTeamMembers } = useTeamStore()
    
    useEffect(() => {
        if (teamId) fetchTeamMembers(teamId)
    }, [teamId])
    
    return { members: teamMembers, isLoading, refetch: () => fetchTeamMembers(teamId) }
}

## 🚨 Common Issues & Fixes

### Issue 1: 403 Forbidden on Create
**Cause:** User doesn't have EVENT_MANAGER or TEAM_LEADER role
**Fix:** Login as a user with proper role, or register as EVENT_MANAGER

### Issue 2: Team not found (404)
**Cause:** Invalid team ID in URL
**Fix:** Check the URL, navigate from teams list

### Issue 3: Can't add member
**Cause:** User already in team or doesn't exist
**Fix:** Check user list, make sure user exists

### Issue 4: Delete fails
**Cause:** Team has events assigned
**Fix:** Delete events first, or implement cascade delete
