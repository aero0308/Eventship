import socketio
from typing import Optional
from uuid import UUID
from app.core.config import settings

# Create Socket.IO server
# cors_allowed_origins should allow frontend URL
sio = socketio.AsyncServer(
    async_mode='asgi',
    cors_allowed_origins=settings.CORS_ORIGINS,
    logger=True,
    engineio_logger=False
)

# Store connected users: {sid: user_id}
connected_users: dict[str, str] = {}

# Store user rooms: {user_id: set of room names}
user_rooms: dict[str, set] = {}


async def authenticate_socket(sid: str, environ: dict) -> Optional[str]:
    """
    Authenticate socket connection using JWT from query params or auth header.
    Returns user_id if valid, None otherwise.
    """
    from jose import jwt, JWTError
    from app.core.config import settings
    
    # Try to get token from query string or auth
    query_string = environ.get('QUERY_STRING', '')
    token = None
    
    if 'token=' in query_string:
        # Parse token from query string
        params = dict(p.split('=') for p in query_string.split('&') if '=' in p)
        token = params.get('token')
    
    if not token:
        # Try Authorization header
        auth_header = environ.get('HTTP_AUTHORIZATION', '')
        if auth_header.startswith('Bearer '):
            token = auth_header[7:]
    
    if not token:
        return None
    
    try:
        payload = jwt.decode(token, settings.JWT_SECRET_KEY, algorithms=[settings.JWT_ALGORITHM])
        return payload.get('sub')
    except JWTError:
        return None


@sio.event
async def connect(sid: str, environ: dict, auth: Optional[dict] = None):
    """Handle client connection"""
    user_id = await authenticate_socket(sid, environ)
    
    if not user_id:
        print(f"❌ Socket {sid} rejected: authentication failed")
        raise socketio.exceptions.ConnectionRefusedError('Authentication failed')
    
    connected_users[sid] = user_id
    user_rooms[user_id] = set()
    
    # Join user's personal room
    await sio.enter_room(sid, f"user_{user_id}")
    user_rooms[user_id].add(f"user_{user_id}")
    
    print(f"✅ User {user_id} connected (sid: {sid})")
    
    # Send confirmation
    await sio.emit('connected', {'status': 'ok', 'user_id': user_id}, to=sid)


@sio.event
async def disconnect(sid: str):
    """Handle client disconnection"""
    user_id = connected_users.pop(sid, None)
    if user_id:
        user_rooms.pop(user_id, None)
        print(f"❌ User {user_id} disconnected (sid: {sid})")


@sio.event
async def join_team(sid: str, data: dict):
    """Join a team room for team-wide updates"""
    user_id = connected_users.get(sid)
    if not user_id:
        return
    
    team_id = data.get('team_id')
    if team_id:
        room = f"team_{team_id}"
        await sio.enter_room(sid, room)
        if user_id in user_rooms:
            user_rooms[user_id].add(room)
        print(f"👥 User {user_id} joined {room}")


@sio.event
async def leave_team(sid: str, data: dict):
    """Leave a team room"""
    team_id = data.get('team_id')
    if team_id:
        room = f"team_{team_id}"
        await sio.leave_room(sid, room)
        user_id = connected_users.get(sid)
        if user_id and user_id in user_rooms:
            user_rooms[user_id].discard(room)


@sio.event
async def join_event(sid: str, data: dict):
    """Join an event room for event-specific updates"""
    user_id = connected_users.get(sid)
    if not user_id:
        return
    
    event_id = data.get('event_id')

        await sio.leave_room(sid, f"event_{event_id}")

async def broadcast_task_updated(task_data: dict, event_id: str, team_id: Optional[str], actor_id: str):
    """
    Broadcast task update to all relevant rooms.
    - To event room (everyone watching event)
    - To team room (team members)
    - To assignee's personal room
    """
    payload = {
        'type': 'task_updated',
        'task': task_data,
        'actor_id': actor_id
    }
    
    # Broadcast to event room
    await sio.emit('task_updated', payload, room=f"event_{event_id}")
    
    # Broadcast to team room if available
    if team_id:
        await sio.emit('task_updated', payload, room=f"team_{team_id}")
    
    # Send to assignee
    if task_data.get('assigned_to'):
        await sio.emit('task_updated', payload, room=f"user_{task_data['assigned_to']}")

import { io, Socket } from 'socket.io-client'
import { ConnectionStatus } from '../types/socket'

class SocketService {
    private socket: Socket | null = null
    private reconnectAttempts = 0
    private maxReconnectAttempts = 5
    private listeners: Map<string, Set<Function>> = new Map()
    private statusListeners: Set<(status: ConnectionStatus) => void> = new Set()
    private currentStatus: ConnectionStatus = 'disconnected'
    
    /**
     * Connect to Socket.IO server with JWT token
     */
    connect(token: string): void {
        if (this.socket?.connected) {
            console.log('Socket already connected')
            return
        }
        
        // Get base URL (strip /api/v1)
        const apiUrl = import.meta.env.VITE_API_URL || 'http://localhost:8000/api/v1'
        const baseUrl = apiUrl.replace('/api/v1', '')
        
        this.setStatus('connecting')
        
        this.socket = io(baseUrl, {
            auth: { token },
            query: { token },
            transports: ['websocket', 'polling'],
            reconnection: true,
            reconnectionAttempts: this.maxReconnectAttempts,
            reconnectionDelay: 1000,
            reconnectionDelayMax: 5000
        })
        
        this.setupListeners()
    }
    
    /**
     * Disconnect from server
     */
    disconnect(): void {
        if (this.socket) {
            this.socket.disconnect()
            this.socket = null
            this.setStatus('disconnected')
            this.reconnectAttempts = 0
        }
    }
    
    /**
     * Register event listeners
     */
    private setupListeners(): void {
        if (!this.socket) return
        
        this.socket.on('connect', () => {
            console.log('✅ Socket connected')
            this.reconnectAttempts = 0
            this.setStatus('connected')
        })
        
        this.socket.on('disconnect', (reason) => {
            console.log('❌ Socket disconnected:', reason)
            this.setStatus('disconnected')
            
            if (reason === 'io server disconnect') {
                // Server initiated disconnect - reconnect manually
                this.socket?.connect()
            }
        })
        
        this.socket.on('connect_error', (error) => {
            console.error('Socket connection error:', error)
            this.reconnectAttempts++
            
            if (this.reconnectAttempts >= this.maxReconnectAttempts) {
                this.setStatus('error')
            } else {
                this.setStatus('reconnecting')
            }
        })
        
        this.socket.io.on('reconnect_attempt', (attempt) => {
            console.log(`Reconnect attempt ${attempt}`)
            this.setStatus('reconnecting')
        })
        
        this.socket.io.on('reconnect', () => {
            console.log('✅ Reconnected')
            this.setStatus('connected')
        })
        
        // Forward all incoming events to registered listeners
        const forwardedEvents = [
            'task_updated',
            'task_created',
            'task_deleted',
            'comment_added',
            'notification',
            'event_updated'
        ]
        
        forwardedEvents.forEach(event => {
            this.socket!.on(event, (data) => {
                this.emit(event, data)
            })
        })
    }
    
    /**
     * Emit event to all registered listeners
     */
    private emit(event: string, data: any): void {
        const listeners = this.listeners.get(event)
        if (listeners) {
            listeners.forEach(callback => {
                try {
                    callback(data)
                } catch (err) {
                    console.error(`Error in listener for ${event}:`, err)
                }
            })
        }
    }
    
    /**
     * Subscribe to an event
     * Returns unsubscribe function
     */
    on(event: string, callback: (data: any) => void): () => void {
        if (!this.listeners.has(event)) {
            this.listeners.set(event, new Set())
        }
        this.listeners.get(event)!.add(callback)
        
        return () => {
            this.listeners.get(event)?.delete(callback)
        }
    }
    
    /**
     * Subscribe to connection status changes
     */
    onStatusChange(callback: (status: ConnectionStatus) => void): () => void {
        this.statusListeners.add(callback)
        callback(this.currentStatus)
        return () => this.statusListeners.delete(callback)
    }
    
    private setStatus(status: ConnectionStatus): void {
        this.currentStatus = status
        this.statusListeners.forEach(cb => cb(status))
    }
    
    /**
     * Join a team room
     */
    joinTeam(teamId: string): void {
        this.socket?.emit('join_team', { team_id: teamId })
    }
    
    leaveTeam(teamId: string): void {
        this.socket?.emit('leave_team', { team_id: teamId })
    }
    
    /**
     * Join an event room
     */
    joinEvent(eventId: string): void {
        this.socket?.emit('join_event', { event_id: eventId })
    }
    
    leaveEvent(eventId: string): void {
        this.socket?.emit('leave_event', { event_id: eventId })
    }
    
    getStatus(): ConnectionStatus {
        return this.currentStatus
    }
    
    isConnected(): boolean {
        return this.socket?.connected || false
    }
}

// Singleton export
export const socketService = new SocketService()

import { useEffect, useState } from 'react'
import { socketService } from '../services/socketService'
import { ConnectionStatus } from '../types/socket'

/**
 * Get current connection status
 */
export const useConnectionStatus = (): ConnectionStatus => {
    const [status, setStatus] = useState<ConnectionStatus>(socketService.getStatus())
    
    useEffect(() => {
        const unsubscribe = socketService.onStatusChange(setStatus)
        return unsubscribe
    }, [])
    
    return status
}

/**
 * Subscribe to a socket event
 */
export const useSocketEvent = <T = any>(
    event: string,
    callback: (data: T) => void,
    deps: any[] = []
): void => {
    useEffect(() => {
        const unsubscribe = socketService.on(event, callback)
        return unsubscribe
    }, deps)
}

import { socketService } from '../services/socketService'
import { useAuthStore } from './authStore'
import toast from 'react-hot-toast'

// Add inside create():

initializeRealtime: () => {
    const currentUserId = useAuthStore.getState().user?.id
    
    // Task updated by someone else
    socketService.on('task_updated', (payload: any) => {
        const { task, actor_id } = payload
        
        // Skip if this user made the change (already updated locally)
        if (actor_id === currentUserId) return
        
        set(state => ({
            tasks: state.tasks.map(t => t.id === task.id ? task : t),
            myTasks: state.myTasks.map(t => t.id === task.id ? task : t),
            currentTask: state.currentTask?.id === task.id ? task : state.currentTask
        }))
        
        // Show toast notification
        const statusText = task.status.replace('_', ' ')
        toast.success(`Task "${task.title}" is now ${statusText}`, {
            icon: '🔄',
            duration: 4000
        })
    })
    
    // Task created by someone else
    socketService.on('task_created', (payload: any) => {
        const { task, actor_id } = payload
        if (actor_id === currentUserId) return
        
        set(state => {
            // Avoid duplicates
            if (state.tasks.some(t => t.id === task.id)) return state
            return { tasks: [task, ...state.tasks] }
        })
        
        toast.success(`New task: "${task.title}"`, {
            icon: '➕',
            duration: 4000
        })
    })
    
    // Task deleted
    socketService.on('task_deleted', (payload: any) => {
        const { task_id, actor_id } = payload
        if (actor_id === currentUserId) return
        
        set(state => ({
            tasks: state.tasks.filter(t => t.id !== task_id),
            myTasks: state.myTasks.filter(t => t.id !== task_id),
            currentTask: state.currentTask?.id === task_id ? null : state.currentTask
        }))
    })
    
    // Comment added
    socketService.on('comment_added', (payload: any) => {
        const { comment, task_id, actor_id } = payload
        if (actor_id === currentUserId) return
        
        // Only add if we're viewing this task
        if (get().currentTask?.id === task_id) {
            set(state => {
                if (state.comments.some(c => c.id === comment.id)) return state
                return { comments: [comment, ...state.comments] }
            })
        }
        
        // Update comment count in tasks list
        set(state => ({
            tasks: state.tasks.map(t => 
                t.id === task_id 
                    ? { ...t, comments_count: t.comments_count + 1 }
                    : t
            )
        }))
    })
}

typescript
import { create } from 'zustand'
import { socketService } from '../services/socketService'
import { useAuthStore } from './authStore'
import toast from 'react-hot-toast'

export interface Notification {
    id: string
    message: string
    type: 'TASK_ASSIGNED' | 'TASK_STATUS_CHANGED' | 'TASK_BLOCKED' | 'COMMENT_ADDED' | 'DEADLINE_APPROACHING'
    read: boolean
    created_at: string
    link?: string
}

interface NotificationState {
    notifications: Notification[]
    unreadCount: number
    addNotification: (notification: Notification) => void
    markAsRead: (id: string) => void
    markAllAsRead: () => void
    removeNotification: (id: string) => void
    clearAll: () => void
    initializeRealtime: () => void
}

export const useNotificationStore = create<NotificationState>((set, get) => ({
    notifications: [],
    unreadCount: 0,
    
    addNotification: (notification) => {
        set(state => ({
            notifications: [notification, ...state.notifications].slice(0, 50),
            unreadCount: state.unreadCount + 1
        }))
    },
    
    markAsRead: (id) => {
        set(state => {
            const notification = state.notifications.find(n => n.id === id)
            if (!notification || notification.read) return state
            
            return {
                notifications: state.notifications.map(n =>
                    n.id === id ? { ...n, read: true } : n
                ),
                unreadCount: Math.max(0, state.unreadCount - 1)
            }
        })
    },
    
    markAllAsRead: () => {
        set(state => ({
            notifications: state.notifications.map(n => ({ ...n, read: true })),
            unreadCount: 0
        }))
    },
    
    removeNotification: (id) => {
        set(state => {
            const notification = state.notifications.find(n => n.id === id)
            const wasUnread = notification && !notification.read
            return {
                notifications: state.notifications.filter(n => n.id !== id),
                unreadCount: wasUnread ? Math.max(0, state.unreadCount - 1) : state.unreadCount
            }
        })
    },
    
    clearAll: () => set({ notifications: [], unreadCount: 0 }),
    
    initializeRealtime: () => {
        const currentUserId = useAuthStore.getState().user?.id
        
        socketService.on('notification', (payload: any) => {
            const { notification } = payload
            
            get().addNotification({
                id: notification.id,
                message: notification.message,
                type: notification.type,
                read: false,
                created_at: notification.created_at
            })
            
            // Show toast
            const icons: Record<string, string> = {
                'TASK_ASSIGNED': '👤',
                'TASK_STATUS_CHANGED': '🔄',
                'TASK_BLOCKED': '🚫',
                'COMMENT_ADDED': '💬',
                'DEADLINE_APPROACHING': '⏰'
            }
            
            toast(notification.message, {
                icon: icons[notification.type] || '🔔',
                duration: 5000
            })
        })
    }
}))

import { useState, useRef, useEffect } from 'react'
import { Link } from 'react-router-dom'
import { useNotificationStore } from '../../stores/notificationStore'
import { formatDistanceToNow } from 'date-fns'

export const NotificationBell = () => {
    const [isOpen, setIsOpen] = useState(false)
    const dropdownRef = useRef<HTMLDivElement>(null)
    const { notifications, unreadCount, markAsRead, markAllAsRead, removeNotification } = useNotificationStore()
    
    useEffect(() => {
        const handleClickOutside = (e: MouseEvent) => {
            if (dropdownRef.current && !dropdownRef.current.contains(e.target as Node)) {
                setIsOpen(false)

        document.addEventListener('mousedown', handleClickOutside)
        return () => document.removeEventListener('mousedown', handleClickOutside)

        'DEADLINE_APPROACHING': '⏰'

                className="relative p-2 text-gray-600 hover:text-gray-900 hover:bg-gray-100 rounded-lg transition"

                <svg className="w-6 h-6" fill="none" viewBox="0 0 24 24" stroke="currentColor">
