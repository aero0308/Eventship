import { useState, useRef, useEffect } from 'react'

interface DropdownItem {
    label: string
    onClick: () => void
    icon?: React.ReactNode
    variant?: 'default' | 'danger'
    disabled?: boolean
}

interface DropdownProps {
    trigger: React.ReactNode
    items: DropdownItem[]
    align?: 'left' | 'right'
}

export const Dropdown = ({ trigger, items, align = 'right' }: DropdownProps) => {
    const [isOpen, setIsOpen] = useState(false)
    const dropdownRef = useRef<HTMLDivElement>(null)
    
    useEffect(() => {
        const handleClickOutside = (e: MouseEvent) => {
            if (dropdownRef.current && !dropdownRef.current.contains(e.target as Node)) {
                setIsOpen(false)
            }
        }
        document.addEventListener('mousedown', handleClickOutside)
        return () => document.removeEventListener('mousedown', handleClickOutside)
    }, [])
    
    return (
        <div className="relative" ref={dropdownRef}>
            <div onClick={() => setIsOpen(!isOpen)}>{trigger}</div>
            {isOpen && (
                <div className={`absolute z-50 mt-2 w-48 bg-white rounded-lg shadow-lg border border-gray-200 py-1 ${align === 'right' ? 'right-0' : 'left-0'}`}>
                    {items.map((item, idx) => (
                        <button
                            key={idx}
                            onClick={() => { item.onClick(); setIsOpen(false) }}
                            disabled={item.disabled}
                            className={`w-full text-left px-4 py-2 text-sm hover:bg-gray-50 disabled:opacity-50 ${item.variant === 'danger' ? 'text-red-600' : 'text-gray-700'}`}
                        >
                            {item.icon && <span className="mr-2">{item.icon}</span>}
                            {item.label}
                        </button>
                    ))}
                </div>
            )}
        </div>
    )
}

import { format, isPast, isToday, isTomorrow } from 'date-fns'

interface TaskCardProps {
    task: Task
    isDragging?: boolean
    onStatusChange?: (status: any) => void
    onEdit?: () => void
    onDelete?: () => void
}

export const TaskCard = ({ 
    task, 
    isDragging, 
    onStatusChange, 
    onEdit, 
    onDelete 
}: TaskCardProps) => {
    const getDueDateColor = () => {
        if (!task.due_date) return 'text-gray-500'
        const dueDate = new Date(task.due_date)
        if (task.status === 'COMPLETED') return 'text-gray-400'
        if (isPast(dueDate) && !isToday(dueDate)) return 'text-red-600 font-semibold'
        if (isToday(dueDate)) return 'text-orange-600 font-semibold'
        if (isTomorrow(dueDate)) return 'text-yellow-600'
        return 'text-gray-500'
    }
    
    const formatDueDate = () => {
        if (!task.due_date) return null
        const dueDate = new Date(task.due_date)
        if (isToday(dueDate)) return 'Due today'
        if (isTomorrow(dueDate)) return 'Due tomorrow'
        if (isPast(dueDate)) return `Overdue: ${format(dueDate, 'MMM dd')}`
        return `Due ${format(dueDate, 'MMM dd')}`
    }
    
    return (
        <div className={`bg-white rounded-lg border border-gray-200 p-4 shadow-sm hover:shadow-md transition-shadow ${isDragging ? 'opacity-50 rotate-2' : ''}`}>
            {/* Header */}
            <div className="flex justify-between items-start gap-2">
                <Link to={`/tasks/${task.id}`} className="flex-1">
                    <h4 className="font-medium text-gray-900 hover:text-blue-600 line-clamp-2">
                        {task.title}
                    </h4>
                </Link>
                {(onEdit || onDelete || onStatusChange) && (
                    <Dropdown
                        trigger={
                            <button className="text-gray-400 hover:text-gray-600 p-1">
                                ⋮
                            </button>
                        }
                        items={[
                            { label: 'View Details', onClick: () => window.location.href = `/tasks/${task.id}` },
                            ...(onEdit ? [{ label: 'Edit', onClick: onEdit }] : []),
                            ...(onDelete ? [{ label: 'Delete', onClick: onDelete, variant: 'danger' as const }] : [])
                        ]}
                    />
                )}
            </div>
            
            {/* Description preview */}
            {task.description && (
                <p className="text-sm text-gray-600 mt-2 line-clamp-2">
                    {task.description}
                </p>
            )}
            
            {/* Event name */}
            {task.event_name && (
                <p className="text-xs text-gray-500 mt-2">
                    📅 {task.event_name}
                </p>
            )}
            
            {/* Badges */}
            <div className="flex flex-wrap gap-2 mt-3">
                <TaskPriorityBadge priority={task.priority} size="sm" />
                {task.is_blocked_by_dependency && (
                    <Badge variant="warning" size="sm">⚠️ Blocked by dependency</Badge>
                )}
            </div>
            
            {/* Footer */}
            <div className="flex items-center justify-between mt-4 pt-3 border-t border-gray-100">
                <div className="flex items-center gap-2">
                    {task.assignee_name ? (
                        <>
                            <Avatar name={task.assignee_name} size="sm" />
                            <span className="text-xs text-gray-600">{task.assignee_name}</span>
                        </>
                    ) : (
                        <span className="text-xs text-gray-400 italic">Unassigned</span>
                    )}
                </div>
                
                {task.due_date && (
                    <span className={`text-xs ${getDueDateColor()}`}>
                        {formatDueDate()}
                    </span>
                )}
            </div>
            
            {/* Counts */}
            <div className="flex gap-3 mt-2 text-xs text-gray-500">
                {task.comments_count > 0 && <span>💬 {task.comments_count}</span>}
                {task.dependencies_count > 0 && <span>🔗 {task.dependencies_count}</span>}
            </div>
        </div>
    )
}

from sqlalchemy.orm import Session, joinedload
from sqlalchemy import func, and_, or_
from typing import List, Optional
from uuid import UUID
from datetime import datetime
from fastapi import HTTPException, status

from app.models.task import Task, TaskStatus, TaskPriority, TaskComment, TaskDependency
from app.models.event import Event
from app.models.user import User
from app.schemas.task import TaskCreate, TaskUpdate, TaskStatusUpdate, CommentCreate

def get_task(db: Session, task_id: UUID) -> Optional[Task]:
    """Get task by ID with all relationships loaded"""
    return db.query(Task).options(
        joinedload(Task.event),
        joinedload(Task.assignee),
        joinedload(Task.creator)
    ).filter(Task.id == task_id).first()

def get_tasks(
    db: Session,
    skip: int = 0,
    limit: int = 100,
    search: Optional[str] = None,
    status: Optional[str] = None,
    priority: Optional[str] = None,
    assigned_to: Optional[UUID] = None,
    event_id: Optional[UUID] = None,
    due_date_from: Optional[datetime] = None,
    due_date_to: Optional[datetime] = None,
    overdue_only: bool = False,
    user: Optional[User] = None
) -> List[Task]:
    """Get tasks with comprehensive filters and authorization"""
    query = db.query(Task).options(
        joinedload(Task.event),
        joinedload(Task.assignee),
        joinedload(Task.creator)
    )
    
    # Authorization based on role
    if user:
        if user.role == "EMPLOYEE":
            # Only tasks assigned to user or in their team
            query = query.filter(Task.assigned_to == user.id)
        elif user.role == "TEAM_LEADER":
            # Tasks in user's team events
            query = query.join(Event).filter(Event.team_id == user.team_id)
        # EVENT_MANAGER: all tasks
    
    # Apply filters
    if search:
        query = query.filter(or_(
            Task.title.ilike(f"%{search}%"),
            Task.description.ilike(f"%{search}%")
        ))
    if status:
        query = query.filter(Task.status == status)
    if priority:
        query = query.filter(Task.priority == priority)
    if assigned_to:
        query = query.filter(Task.assigned_to == assigned_to)
    if event_id:
        query = query.filter(Task.event_id == event_id)
    if due_date_from:
        query = query.filter(Task.due_date >= due_date_from)
    if due_date_to:
        query = query.filter(Task.due_date <= due_date_to)
    if overdue_only:
        query = query.filter(
            Task.due_date < datetime.utcnow(),
            Task.status != TaskStatus.COMPLETED
        )
    
    return query.order_by(Task.created_at.desc()).offset(skip).limit(limit).all()

def get_my_tasks(
    db: Session,
    user_id: UUID,
    status: Optional[str] = None
) -> List[Task]:
    """Get tasks assigned to current user"""
    query = db.query(Task).options(
        joinedload(Task.event),
        joinedload(Task.creator)
    ).filter(Task.assigned_to == user_id)
    
    if status:
        query = query.filter(Task.status == status)
    
    return query.order_by(Task.due_date.asc().nullslast()).all()

def create_task(db: Session, task_data: TaskCreate, user_id: UUID) -> Task:
    """Create new task"""
    # Verify event exists
    event = db.query(Event).filter(Event.id == task_data.event_id).first()
    if not event:
        raise HTTPException(status_code=404, detail="Event not found")
    
    # Verify assignee exists if provided
    if task_data.assigned_to:
        assignee = db.query(User).filter(User.id == task_data.assigned_to).first()
        if not assignee:
            raise HTTPException(status_code=404, detail="Assignee not found")
    
    task = Task(
        **task_data.dict(),
        created_by=user_id
    )
    db.add(task)
    db.commit()
    db.refresh(task)
    return task

def update_task(db: Session, task_id: UUID, task_data: TaskUpdate) -> Task:
    """Update task"""
    task = get_task(db, task_id)
    if not task:
        raise HTTPException(status_code=404, detail="Task not found")
    
    update_dict = task_data.dict(exclude_unset=True)
    for key, value in update_dict.items():
        setattr(task, key, value)
    
    task.updated_at = datetime.utcnow()
    db.commit()
    db.refresh(task)
    return task

def delete_task(db: Session, task_id: UUID) -> bool:
    """Delete task"""
    task = get_task(db, task_id)
    if not task:
        raise HTTPException(status_code=404, detail="Task not found")
    
    db.delete(task)
    db.commit()
    return True

def update_task_status(
    db: Session,
    task_id: UUID,
    status_data: TaskStatusUpdate,
    user_id: UUID
) -> Task:
    """Update task status and optionally add a comment"""
    task = get_task(db, task_id)
    if not task:
        raise HTTPException(status_code=404, detail="Task not found")
    
    # If marking as completed, check dependencies
    if status_data.status == TaskStatus.COMPLETED:
        incomplete_deps = db.query(TaskDependency).join(
            Task, TaskDependency.depends_on_task_id == Task.id
        ).filter(
            TaskDependency.task_id == task_id,
            Task.status != TaskStatus.COMPLETED
        ).count()
        
        if incomplete_deps > 0:
            raise HTTPException(
                status_code=400,
                detail=f"Cannot complete task. {incomplete_deps} dependencies are not completed."
            )
    
    task.status = status_data.status
    task.updated_at = datetime.utcnow()
    
    # Add comment if provided (for blocker reporting)
    if status_data.comment:
        comment = TaskComment(
            content=status_data.comment,
            task_id=task_id,
            user_id=user_id
        )
        db.add(comment)
    
    db.commit()
    db.refresh(task)
    return task

def assign_task(db: Session, task_id: UUID, assignee_id: UUID) -> Task:
    """Assign or reassign task"""
    task = get_task(db, task_id)
    if not task:
        raise HTTPException(status_code=404, detail="Task not found")
    
    assignee = db.query(User).filter(User.id == assignee_id).first()
    if not assignee:
        raise HTTPException(status_code=404, detail="User not found")
    
    task.assigned_to = assignee_id
    task.updated_at = datetime.utcnow()
    db.commit()
    db.refresh(task)
    return task

def get_task_comments(db: Session, task_id: UUID) -> List[TaskComment]:
    """Get all comments for a task"""
    return db.query(TaskComment).options(
        joinedload(TaskComment.user)
    ).filter(
        TaskComment.task_id == task_id
    ).order_by(TaskComment.created_at.desc()).all()

def add_comment(
    db: Session,

    """Add comment to task (used for reporting blockers)"""
    task = get_task(db, task_id)
    if not task:
        raise HTTPException(status_code=404, detail="Task not found")
    
    comment = TaskComment(
        content=comment_data.content,

    db.add(comment)
    db.commit()

def delete_comment(db: Session, comment_id: UUID, user_id: UUID) -> bool:
    """Delete comment (only owner)"""
    comment = db.query(TaskComment).filter(TaskComment.id == comment_id).first()
    if not comment:
        raise HTTPException(status_code=404, detail="Comment not found")
    
    if comment.user_id != user_id:
        raise HTTPException(status_code=403, detail="Not authorized")
    
    db.delete(comment)
    db.commit()
    return True

def get_task_dependencies(db: Session, task_id: UUID) -> List[TaskDependency]:
    """Get dependencies for a task"""
    return db.query(TaskDependency).options(
        joinedload(TaskDependency.depends_on_task)
    ).filter(TaskDependency.task_id == task_id).all()

def add_dependency(
    db: Session,
    task_id: UUID,
    depends_on_task_id: UUID
) -> TaskDependency:
    """Add dependency (task depends on another task)"""
    if task_id == depends_on_task_id:
        raise HTTPException(status_code=400, detail="Task cannot depend on itself")
    
    # Check both tasks exist
    task = get_task(db, task_id)
    depends_on = get_task(db, depends_on_task_id)
    if not task or not depends_on:
        raise HTTPException(status_code=404, detail="Task not found")
    
    # Check for circular dependency
    if _has_circular_dependency(db, task_id, depends_on_task_id):
        raise HTTPException(status_code=400, detail="Circular dependency detected")
    
    # Check if dependency already exists
    existing = db.query(TaskDependency).filter(
        TaskDependency.task_id == task_id,
        TaskDependency.depends_on_task_id == depends_on_task_id
    ).first()
    if existing:
        raise HTTPException(status_code=400, detail="Dependency already exists")
    
    dependency = TaskDependency(
        task_id=task_id,
        depends_on_task_id=depends_on_task_id
    )
    db.add(dependency)
    db.commit()
    db.refresh(dependency)
    return dependency

def _has_circular_dependency(db: Session, task_id: UUID, depends_on_id: UUID) -> bool:
    """Check if adding this dependency creates a cycle"""
    visited = set()
    to_visit = [depends_on_id]
    
    while to_visit:
        current = to_visit.pop()
        if current == task_id:
            return True
        if current in visited:
            continue
        visited.add(current)
        
        deps = db.query(TaskDependency).filter(
            TaskDependency.task_id == current
        ).all()
        to_visit.extend([d.depends_on_task_id for d in deps])
    
    return False

def remove_dependency(
    db: Session,
    task_id: UUID,
    depends_on_task_id: UUID
) -> bool:
    """Remove dependency"""
    dep = db.query(TaskDependency).filter(
        TaskDependency.task_id == task_id,
        TaskDependency.depends_on_task_id == depends_on_task_id
    ).first()
    
    if not dep:
        raise HTTPException(status_code=404, detail="Dependency not found")
    
    db.delete(dep)
    db.commit()
    return True

def get_task_stats(db: Session, user: Optional[User] = None) -> dict:
    """Get overall task statistics"""
    query = db.query(Task)
    
    if user and user.role == "EMPLOYEE":
        query = query.filter(Task.assigned_to == user.id)
    
    total = query.count()
    not_started = query.filter(Task.status == TaskStatus.NOT_STARTED).count()
    in_progress = query.filter(Task.status == TaskStatus.IN_PROGRESS).count()
    blocked = query.filter(Task.status == TaskStatus.BLOCKED).count()
    completed = query.filter(Task.status == TaskStatus.COMPLETED).count()
    overdue = query.filter(
        Task.due_date < datetime.utcnow(),
        Task.status != TaskStatus.COMPLETED
    ).count()
    
    completion_rate = (completed / total * 100) if total > 0 else 0
    
    return {
        "total": total,
        "not_started": not_started,
        "in_progress": in_progress,
        "blocked": blocked,
        "completed": completed,
        "overdue": overdue,
        "completion_rate": round(completion_rate, 2)
    }

    onTaskStatusChange: (taskId: string, status: TaskStatus) => void
}

const columnConfig: Record<TaskStatus, { title: string; color: string; bgColor: string }> = {
    NOT_STARTED: { title: 'Not Started', color: 'text-gray-700', bgColor: 'bg-gray-50' },

export const TaskColumn = ({ status, tasks, onTaskStatusChange }: TaskColumnProps) => {

    return (
        <div

import { useSortable } from '@dnd-kit/sortable'
import { CSS } from '@dnd-kit/utilities'
import { Task, TaskStatus } from '../../types/task'
import { TaskCard } from './TaskCard'

interface DraggableTaskCardProps {
    task: Task
    onStatusChange: (status: TaskStatus) => void
}

export const DraggableTaskCard = ({ task, onStatusChange }: DraggableTaskCardProps) => {
    const {
        attributes,
        listeners,
        setNodeRef,
        transform,
        transition,
        isDragging
    } = useSortable({ 
        id: task.id,
        data: { task }
    })
    
    const style = {
        transform: CSS.Transform.toString(transform),
        transition,
        cursor: isDragging ? 'grabbing' : 'grab'
    }
    
    return (
        <div
            ref={setNodeRef}
            style={style}
            {...attributes}
            {...listeners}
        >
            <TaskCard task={task} isDragging={isDragging} />
        </div>
    )
}

import { useState } from 'react'
import {
    DndContext,
    DragEndEvent,
    DragOverlay,
    DragStartEvent,
    PointerSensor,
    useSensor,
    useSensors,
    closestCorners
} from '@dnd-kit/core'
import { Task, TaskStatus } from '../../types/task'
import { useTaskStore } from '../../stores/taskStore'
import { TaskColumn } from './TaskColumn'
import { TaskCard } from './TaskCard'
import toast from 'react-hot-toast'

interface TaskBoardProps {
    tasks: Task[]
    onTaskClick?: (task: Task) => void
}

export const TaskBoard = ({ tasks, onTaskClick }: TaskBoardProps) => {
    const [activeTask, setActiveTask] = useState<Task | null>(null)
    const { updateStatus } = useTaskStore()
    
    const sensors = useSensors(
        useSensor(PointerSensor, {
            activationConstraint: {
                distance: 8 // Prevent accidental drags
            }
        })
    )
    
    const tasksByStatus: Record<TaskStatus, Task[]> = {
        NOT_STARTED: tasks.filter(t => t.status === 'NOT_STARTED'),
        IN_PROGRESS: tasks.filter(t => t.status === 'IN_PROGRESS'),
        BLOCKED: tasks.filter(t => t.status === 'BLOCKED'),
        COMPLETED: tasks.filter(t => t.status === 'COMPLETED')
    }
    
    const handleDragStart = (event: DragStartEvent) => {
        const task = tasks.find(t => t.id === event.active.id)
        setActiveTask(task || null)
    }
    
    const handleDragEnd = async (event: DragEndEvent) => {
        setActiveTask(null)
        
        const { active, over } = event
        if (!over) return
        
        const taskId = active.id as string
        const task = tasks.find(t => t.id === taskId)
        if (!task) return
        
        // Determine new status
        let newStatus: TaskStatus
        
        // If dropped on a column
        if (['NOT_STARTED', 'IN_PROGRESS', 'BLOCKED', 'COMPLETED'].includes(over.id as string)) {
            newStatus = over.id as TaskStatus
        } else {
            // If dropped on another task, get that task's status
            const overTask = tasks.find(t => t.id === over.id)
            if (!overTask) return
            newStatus = overTask.status
        }
        
        // If status didn't change, do nothing
        if (task.status === newStatus) return
        
        // If moving to BLOCKED, ask for reason
        let comment: string | undefined
        if (newStatus === 'BLOCKED') {
            comment = window.prompt('Why is this task blocked?') || undefined
            if (!comment) return // Cancelled
        }
        
        // If moving to COMPLETED, check for incomplete dependencies
        if (newStatus === 'COMPLETED' && task.is_blocked_by_dependency) {
            toast.error('Cannot complete task with incomplete dependencies')
            return
        }
        
        try {
            await updateStatus(taskId, { status: newStatus, comment })
            
            // Show toast
            if (newStatus === 'BLOCKED') {
                toast.warning('Task marked as blocked')
            } else if (newStatus === 'COMPLETED') {
                toast.success('Task completed!')
            } else {
                toast.success(`Task moved to ${newStatus.replace('_', ' ')}`)
            }
        } catch (error: any) {
            toast.error(error.response?.data?.detail || 'Failed to update task')
        }
    }
    
    return (
        <DndContext
            sensors={sensors}
            collisionDetection={closestCorners}
            onDragStart={handleDragStart}
            onDragEnd={handleDragEnd}
        >
            <div className="flex gap-4 overflow-x-auto pb-4">
                <TaskColumn 
                    status="NOT_STARTED" 
                    tasks={tasksByStatus.NOT_STARTED}
                    onTaskStatusChange={(id, s) => updateStatus(id, { status: s })}
                />
                <TaskColumn 
                    status="IN_PROGRESS" 
                    tasks={tasksByStatus.IN_PROGRESS}
                    onTaskStatusChange={(id, s) => updateStatus(id, { status: s })}
                />
                <TaskColumn 
                    status="BLOCKED" 
                    tasks={tasksByStatus.BLOCKED}
                    onTaskStatusChange={(id, s) => updateStatus(id, { status: s })}
                />
                <TaskColumn 
                    status="COMPLETED" 
                    tasks={tasksByStatus.COMPLETED}
                    onTaskStatusChange={(id, s) => updateStatus(id, { status: s })}
                />
            </div>
            
            {/* Drag overlay - shows the card being dragged */}
            <DragOverlay>
                {activeTask ? (
                    <div className="rotate-3 opacity-90">
                        <TaskCard task={activeTask} isDragging />
                    </div>
                ) : null}
            </DragOverlay>
        </DndContext>
    )
}

interface TaskCommentsProps {
    taskId: string
}

export const TaskComments = ({ taskId }: TaskCommentsProps) => {
    const { user } = useAuth()
    const { comments, addComment, deleteComment } = useTaskStore()

        <div className="space-y-4">
            <h3 className="font-semibold">Comments ({comments.length})</h3>
            
            {/* Add comment form */}

import { useState } from 'react'
import { useTaskStore } from '../../stores/taskStore'
import { Task } from '../../types/task'
import { TaskStatusBadge } from './TaskStatusBadge'
import { Button } from '../ui/Button'
import { Select } from '../ui/Select'
import { Modal } from '../ui/Modal'
import { Link } from 'react-router-dom'
import toast from 'react-hot-toast'

interface TaskDependenciesProps {
    taskId: string
    allTasks: Task[]
}

export const TaskDependencies = ({ taskId, allTasks }: TaskDependenciesProps) => {
    const { dependencies, addDependency, removeDependency } = useTaskStore()
    const [isModalOpen, setIsModalOpen] = useState(false)
    const [selectedTaskId, setSelectedTaskId] = useState('')
    const [isSubmitting, setIsSubmitting] = useState(false)
    
    // Filter out tasks that already depend on this task
    const availableTasks = allTasks.filter(t => 
        t.id !== taskId && 
        !dependencies.some(d => d.depends_on_task_id === t.id)
    )
    
    const handleAdd = async () => {
        if (!selectedTaskId) return
        setIsSubmitting(true)
        try {
            await addDependency(taskId, selectedTaskId)
            toast.success('Dependency added')
            setIsModalOpen(false)
            setSelectedTaskId('')
        } catch (error: any) {
            toast.error(error.response?.data?.detail || 'Failed to add dependency')
        } finally {
            setIsSubmitting(false)
        }
    }
    
    const handleRemove = async (dependsOnId: string) => {
        if (!confirm('Remove this dependency?')) return
        try {
            await removeDependency(taskId, dependsOnId)
            toast.success('Dependency removed')
        } catch (error) {
            toast.error('Failed to remove dependency')
        }
    }
    
    return (
        <div className="space-y-4">
            <div className="flex justify-between items-center">
                <h3 className="font-semibold">Dependencies ({dependencies.length})</h3>
                <Button size="sm" onClick={() => setIsModalOpen(true)}>
                    Add Dependency
                </Button>
            </div>
            
            {/* Info message */}

            {/* Dependencies list */}

                        <div className="flex items-center gap-3 flex-1">
                            <Link 
                                to={`/tasks/${dep.depends_on_task_id}`}
                                className="font-medium text-sm hover:text-blue-600"
                            >
                                {dep.depends_on_task_title}
                            </Link>
                            <TaskStatusBadge status={dep.depends_on_task_status} size="sm" />
                        </div>
                        <button
                            onClick={() => handleRemove(dep.depends_on_task_id)}
                            className="text-red-500 hover:text-red-700 text-sm"
                        >
                            Remove
                        </button>

                {dependencies.length === 0 && (

import { useState } from 'react'
import { TaskStatus } from '../../types/task'
import { Modal } from '../ui/Modal'
import { Textarea } from '../ui/Textarea'
import { Button } from '../ui/Button'

interface TaskStatusUpdateModalProps {
    isOpen: boolean
    onClose: () => void
    currentStatus: TaskStatus
    newStatus: TaskStatus | null
    onSubmit: (comment?: string) => Promise<void>
}

const statusMessages: Record<TaskStatus, string> = {
    NOT_STARTED: 'Why is this task not started?',
    IN_PROGRESS: 'Add a progress note (optional)',
    BLOCKED: 'Why is this task blocked? (required)',
    COMPLETED: 'Add a completion note (optional)'
}

export const TaskStatusUpdateModal = ({
    isOpen,
    onClose,
    currentStatus,
    newStatus,
    onSubmit
}: TaskStatusUpdateModalProps) => {
    const [comment, setComment] = useState('')
    const [isSubmitting, setIsSubmitting] = useState(false)
    
    if (!newStatus) return null
    
    const isBlockedStatus = newStatus === 'BLOCKED'
    const isRequired = isBlockedStatus
    
    const handleSubmit = async () => {
        if (isRequired && !comment.trim()) return
        
        setIsSubmitting(true)
        try {
            await onSubmit(comment.trim() || undefined)
            setComment('')
            onClose()
        } finally {
            setIsSubmitting(false)
        }
    }
    
    return (
        <Modal isOpen={isOpen} onClose={onClose} title="Update Task Status">
            <div className="space-y-4">
                <p className="text-sm text-gray-600">
                    Moving from <strong>{currentStatus.replace('_', ' ')}</strong> to{' '}
                    <strong>{newStatus.replace('_', ' ')}</strong>
                </p>
                
                <Textarea
                    label={statusMessages[newStatus]}
                    value={comment}
                    onChange={(e) => setComment(e.target.value)}
                    placeholder={isBlockedStatus 
                        ? "e.g., Waiting for venue confirmation, budget not approved, etc." 
                        : "Add optional notes..."
                    }
                    rows={4}
                    error={isRequired && !comment.trim() ? 'Reason is required for blocked tasks' : undefined}
                />
                
                <div className="flex gap-2 justify-end">
                    <Button variant="secondary" onClick={onClose}>
                        Cancel
                    </Button>
                    <Button 
                        onClick={handleSubmit} 
                        isLoading={isSubmitting}
                        disabled={isRequired && !comment.trim()}
                    >
                        Update Status
                    </Button>
                </div>
            </div>
        </Modal>
    )
}

import { useForm } from 'react-hook-form'
import { zodResolver } from '@hookform/resolvers/zod'
import { z } from 'zod'
import { useState, useEffect } from 'react'
import { Task, TaskCreate, TaskPriority, TaskStatus } from '../../types/task'
import { Event } from '../../types/event'
import { Team } from '../../types/team'
import { User } from '../../types/user'
import { Input } from '../ui/Input'
import { Textarea } from '../ui/Textarea'
import { Select } from '../ui/Select'
import { DatePicker } from '../ui/DatePicker'
import { Button } from '../ui/Button'
import api from '../../services/api'

const taskSchema = z.object({
    title: z.string().min(3, 'Title must be at least 3 characters').max(200),
    description: z.string().max(2000).optional(),
    priority: z.enum(['HIGH', 'MEDIUM', 'LOW']),
    status: z.enum(['NOT_STARTED', 'IN_PROGRESS', 'BLOCKED', 'COMPLETED']).optional(),
    event_id: z.string().min(1, 'Event is required'),
    assigned_to: z.string().optional(),
    due_date: z.string().optional(),
    estimated_hours: z.number().min(0).optional()
})

interface TaskFormProps {
    task?: Task
    events: Event[]
    teams: Team[]
    onSubmit: (data: TaskCreate) => Promise<void>
    isLoading?: boolean
}

export const TaskForm = ({ task, events, teams, onSubmit, isLoading }: TaskFormProps) => {
    const [users, setUsers] = useState<User[]>([])
    const { register, handleSubmit, watch, formState: { errors } } = useForm({
        resolver: zodResolver(taskSchema),
        defaultValues: task ? {
            title: task.title,
            description: task.description || '',
            priority: task.priority,
            status: task.status,
            event_id: task.event_id,
            assigned_to: task.assigned_to || '',
            due_date: task.due_date ? task.due_date.slice(0, 16) : '',
            estimated_hours: task.estimated_hours || undefined
        } : {
            priority: 'MEDIUM' as TaskPriority,
            status: 'NOT_STARTED' as TaskStatus
        }
    })
    
    const selectedEventId = watch('event_id')
    
    // Fetch users when event selected (users in event's team)
    useEffect(() => {
        if (selectedEventId) {
            const event = events.find(e => e.id === selectedEventId)
            if (event?.team_id) {
                api.get('/users', { params: { team_id: event.team_id } })
                    .then(res => setUsers(res.data))
                    .catch(() => setUsers([]))
            }
        }
    }, [selectedEventId, events])
    
    return (
        <form onSubmit={handleSubmit((data) => onSubmit(data as TaskCreate))} className="space-y-6">
            <Input
                label="Task Title"
                placeholder="e.g., Book conference venue"
                error={errors.title?.message}
                {...register('title')}
            />
            
            <Textarea
                label="Description"
                placeholder="Describe what needs to be done..."
                error={errors.description?.message}
                rows={4}
                {...register('description')}
            />
            
            <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
                <Select
                    label="Event"
                    error={errors.event_id?.message}
                    placeholder="Select event..."
                    options={events.map(e => ({ value: e.id, label: e.name }))}
                    {...register('event_id')}
                />
                
                <Select
                    label="Assign To"
                    error={errors.assigned_to?.message}
                    placeholder="Unassigned"
                    options={[
                        { value: '', label: 'Unassigned' },
                        ...users.map(u => ({ value: u.id, label: u.full_name }))
                    ]}
                    {...register('assigned_to')}
                />
            </div>
            
            <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
                <Select
                    label="Priority"
                    error={errors.priority?.message}
                    options={[
                        { value: 'HIGH', label: '🔴 High' },
                        { value: 'MEDIUM', label: '🟡 Medium' },
                        { value: 'LOW', label: '🟢 Low' }
                    ]}
                    {...register('priority')}
                />

                    <Select

                            { value: 'NOT_STARTED', label: 'Not Started' },
                            { value: 'IN_PROGRESS', label: 'In Progress' },
